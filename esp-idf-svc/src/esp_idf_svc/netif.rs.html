1//! Network abstraction
2//!
3//! The purpose of ESP-NETIF library is twofold:
4//!
5//! - It provides an abstraction layer for the application on top of the TCP/IP
6//!   stack. This will allow applications to choose between IP stacks in the
7//!   future.
8//! - The APIs it provides are thread safe, even if the underlying TCP/IP
9//!   stack APIs are not.

45    Slip,
46    #[cfg(all(esp_idf_comp_openthread_enabled, esp_idf_openthread_enabled,))]

48    /// Layer 2 bridge between other netifs (see [`EspNetifBridge`])
49    #[cfg(all(esp_idf_esp_netif_tcpip_lwip, esp_idf_esp_netif_bridge_en))]
50    Bridge(BridgeConfiguration),
51}

101    pub fn default_configuration(&self) -> NetifConfiguration {
102        match self {
103            Self::Sta => NetifConfiguration::wifi_default_client(),
104            #[cfg(esp_idf_esp_wifi_softap_support)]
105            Self::Ap => NetifConfiguration::wifi_default_router(),
106            Self::Eth => NetifConfiguration::eth_default_client(),
107            #[cfg(esp_idf_lwip_ppp_support)]
108            Self::Ppp => NetifConfiguration::ppp_default_client(),
109            #[cfg(esp_idf_lwip_slip_support)]
110            Self::Slip => NetifConfiguration::slip_default_client(),
111            #[cfg(all(esp_idf_comp_openthread_enabled, esp_idf_openthread_enabled,))]

113            #[cfg(all(esp_idf_esp_netif_tcpip_lwip, esp_idf_esp_netif_bridge_en))]
114            Self::Bridge(_) => NetifConfiguration {
115                stack: *self,
116                ..NetifConfiguration::bridge_default()

121    fn default_mac(&self) -> Result<Option<[u8; 6]>, EspError> {
122        if let Some(mac_type) = self.default_mac_raw_type() {
123            let mut mac = [0; 6];
124
125            let result = esp!(unsafe { esp_read_mac(mac.as_mut_ptr() as *mut _, mac_type) });
126
127            // On SoCs without native WiFi hardware (e.g., ESP32-P4 using
128            // esp_wifi_remote), the eFuse has no WiFi MAC address.
129            // esp_read_mac will fail for WiFi MAC types. Return None and let
130            // esp_wifi_start() set the real MAC from the companion chip later.
131            #[cfg(not(esp_idf_soc_wifi_supported))]
132            {
133                if result.is_ok() {
134                    Ok(Some(mac))
135                } else {
136                    Ok(None)
137                }
138            }
139
140            #[cfg(esp_idf_soc_wifi_supported)]
141            {
142                result?;
143
144                Ok(Some(mac))
145            }
146        } else {
147            Ok(None)
148        }
149    }

151    fn default_mac_raw_type(&self) -> Option<esp_mac_type_t> {
152        match self {
153            Self::Sta => Some(esp_mac_type_t_ESP_MAC_WIFI_STA),
154            #[cfg(esp_idf_esp_wifi_softap_support)]
155            Self::Ap => Some(esp_mac_type_t_ESP_MAC_WIFI_SOFTAP),
156            Self::Eth => Some(esp_mac_type_t_ESP_MAC_ETH),
157            // The bridge shares the MAC address of its Ethernet port(s)
158            #[cfg(all(esp_idf_esp_netif_tcpip_lwip, esp_idf_esp_netif_bridge_en))]
159            Self::Bridge(_) => Some(esp_mac_type_t_ESP_MAC_ETH),
160            #[cfg(all(esp_idf_comp_openthread_enabled, esp_idf_openthread_enabled,))]
161            Self::Thread => {
162                #[cfg(esp_idf_soc_ieee802154_supported)]
163                let mac_type = Some(esp_mac_type_t_ESP_MAC_IEEE802154);
164
165                #[cfg(not(esp_idf_soc_ieee802154_supported))]
166                let mac_type = None;
167
168                mac_type
169            }
170            #[cfg(any(esp_idf_lwip_slip_support, esp_idf_lwip_ppp_support))]
171            _ => None,
172        }
173    }

175    fn default_raw_stack(&self) -> *const esp_netif_netstack_config_t {
176        unsafe {
177            match self {
178                Self::Sta => _g_esp_netif_netstack_default_wifi_sta,
179                #[cfg(esp_idf_esp_wifi_softap_support)]
180                Self::Ap => _g_esp_netif_netstack_default_wifi_ap,
181                Self::Eth => _g_esp_netif_netstack_default_eth,
182                #[cfg(esp_idf_lwip_ppp_support)]
183                Self::Ppp => _g_esp_netif_netstack_default_ppp,
184                #[cfg(esp_idf_lwip_slip_support)]
185                Self::Slip => _g_esp_netif_netstack_default_slip,
186                #[cfg(all(
187                    esp_idf_comp_openthread_enabled,
188                    esp_idf_openthread_enabled,
189                    esp_idf_version_major = "4"
190                ))]
191                Self::Thread => _g_esp_netif_netstack_default_openthread,
192                #[cfg(all(
193                    esp_idf_comp_openthread_enabled,
194                    esp_idf_openthread_enabled,
195                    not(esp_idf_version_major = "4")
196                ))]
197                Self::Thread => &g_esp_netif_netstack_default_openthread,
198                #[cfg(all(esp_idf_esp_netif_tcpip_lwip, esp_idf_esp_netif_bridge_en))]
199                Self::Bridge(_) => _g_esp_netif_netstack_default_br,
200            }
201        }
202    }

410/// Return `conf` as the default IP configuration of a netif, or `None` without IPv4, as
411/// ESP-IDF then has neither a DHCP client / server nor static IPv4 addressing.
412// `CONFIG_LWIP_IPV4` only exists since ESP-IDF 5.1; before that IPv4 is always enabled
413fn default_ip_configuration(conf: ipv4::Configuration) -> Option<ipv4::Configuration> {
414    cfg!(any(esp_idf_lwip_ipv4, not(esp_idf_version_at_least_5_1_0))).then_some(conf)
415}
416
417static INITALIZED: mutex::Mutex<bool> = mutex::Mutex::new(false);
418
419fn initialize_netif_stack() -> Result<(), EspError> {
420    let mut guard = INITALIZED.lock();
421
422    if !*guard {
423        esp!(unsafe { esp_netif_init() })?;
424
425        *guard = true;
426    }
427
428    Ok(())
429}

443    #[allow(unused_mut)]
444    pub fn new_with_conf(conf: &NetifConfiguration) -> Result<Self, EspError> {
445        // Without IPv4, ESP-IDF has neither a DHCP client / server nor static IPv4 addressing
446        #[cfg(not(any(esp_idf_lwip_ipv4, not(esp_idf_version_at_least_5_1_0))))]
447        if conf.ip_configuration.is_some() {
448            return Err(EspError::from_infallible::<ESP_ERR_NOT_SUPPORTED>());
449        }
450
451        initialize_netif_stack()?;
452
453        let c_if_key = to_cstring_arg(conf.key.as_str())?;
454        let c_if_description = to_cstring_arg(conf.description.as_str())?;
455
456        let initial_mac = if let Some(custom_mac) = conf.custom_mac {
457            custom_mac
458        } else {
459            conf.stack.default_mac()?.unwrap_or([0; 6])
460        };
461
462        let (mut esp_inherent_config, ip_info, dhcps, dns, secondary_dns, hostname) = match conf
463            .ip_configuration
464        {
465            Some(ipv4::Configuration::Client(ref ip_conf)) => (
466                esp_netif_inherent_config_t {
467                    flags: conf.flags
468                        | (if matches!(ip_conf, ipv4::ClientConfiguration::DHCP(_)) {
469                            esp_netif_flags_ESP_NETIF_DHCP_CLIENT
470                        } else {
471                            0
472                        }),
473                    mac: initial_mac,
474                    ip_info: ptr::null(),
475                    get_ip_event: conf.got_ip_event_id.map(NonZeroU32::get).unwrap_or(0),
476                    lost_ip_event: conf.lost_ip_event_id.map(NonZeroU32::get).unwrap_or(0),
477                    if_key: c_if_key.as_c_str().as_ptr() as _,
478                    if_desc: c_if_description.as_c_str().as_ptr() as _,
479                    route_prio: conf.route_priority as _,
480                    #[cfg(not(esp_idf_version_major = "4"))]
481                    bridge_info: ptr::null_mut(),
482                    #[cfg(esp_idf_version_at_least_6_0_0)]
483                    mtu: 0,
484                },
485                match ip_conf {
486                    ipv4::ClientConfiguration::DHCP(_) => None,
487                    ipv4::ClientConfiguration::Fixed(ref fixed_conf) => Some(esp_netif_ip_info_t {
488                        ip: Newtype::<esp_ip4_addr_t>::from(fixed_conf.ip).0,
489                        netmask: Newtype::<esp_ip4_addr_t>::from(fixed_conf.subnet.mask).0,
490                        gw: Newtype::<esp_ip4_addr_t>::from(fixed_conf.subnet.gateway).0,
491                    }),
492                },
493                false,
494                match ip_conf {
495                    ipv4::ClientConfiguration::DHCP(_) => None,
496                    ipv4::ClientConfiguration::Fixed(ref fixed_conf) => fixed_conf.dns,
497                },
498                match ip_conf {
499                    ipv4::ClientConfiguration::DHCP(_) => None,
500                    ipv4::ClientConfiguration::Fixed(ref fixed_conf) => fixed_conf.secondary_dns,
501                },
502                match ip_conf {
503                    ipv4::ClientConfiguration::DHCP(ref dhcp_conf) => dhcp_conf.hostname.as_ref(),
504                    ipv4::ClientConfiguration::Fixed(_) => None,
505                },
506            ),
507            Some(ipv4::Configuration::Router(ref ip_conf)) => (
508                esp_netif_inherent_config_t {
509                    flags: conf.flags
510                        | (if ip_conf.dhcp_enabled {
511                            esp_netif_flags_ESP_NETIF_DHCP_SERVER
512                        } else {
513                            0
514                        })
515                        | esp_netif_flags_ESP_NETIF_FLAG_AUTOUP,
516                    mac: initial_mac,
517                    ip_info: ptr::null(),
518                    get_ip_event: conf.got_ip_event_id.map(NonZeroU32::get).unwrap_or(0),
519                    lost_ip_event: conf.lost_ip_event_id.map(NonZeroU32::get).unwrap_or(0),
520                    if_key: c_if_key.as_c_str().as_ptr() as _,
521                    if_desc: c_if_description.as_c_str().as_ptr() as _,
522                    route_prio: conf.route_priority as _,
523                    #[cfg(not(esp_idf_version_major = "4"))]
524                    bridge_info: ptr::null_mut(),
525                    #[cfg(esp_idf_version_at_least_6_0_0)]
526                    mtu: 0,
527                },
528                Some(esp_netif_ip_info_t {
529                    ip: Newtype::<esp_ip4_addr_t>::from(ip_conf.subnet.gateway).0,
530                    netmask: Newtype::<esp_ip4_addr_t>::from(ip_conf.subnet.mask).0,
531                    gw: Newtype::<esp_ip4_addr_t>::from(ip_conf.subnet.gateway).0,
532                }),
533                ip_conf.dhcp_enabled,
534                ip_conf.dns,
535                None, /* For APs, ESP-IDF supports setting a primary DNS only ip_conf.secondary_dns */
536                None,
537            ),
538            None => {
539                let mut flags = conf.flags | esp_netif_flags_ESP_NETIF_FLAG_AUTOUP;
540
541                #[cfg(esp_idf_lwip_ppp_support)]
542                {
543                    if matches!(conf.stack, NetifStack::Ppp) {
544                        flags &= !esp_netif_flags_ESP_NETIF_FLAG_AUTOUP;
545                    }
546                }
547
548                (
549                    esp_netif_inherent_config_t {
550                        flags,
551                        mac: initial_mac,
552                        ip_info: ptr::null(),
553                        get_ip_event: conf.got_ip_event_id.map(NonZeroU32::get).unwrap_or(0),
554                        lost_ip_event: conf.lost_ip_event_id.map(NonZeroU32::get).unwrap_or(0),
555                        if_key: c_if_key.as_c_str().as_ptr() as _,
556                        if_desc: c_if_description.as_c_str().as_ptr() as _,
557                        route_prio: conf.route_priority as _,
558                        #[cfg(not(esp_idf_version_major = "4"))]
559                        bridge_info: ptr::null_mut(),
560                        #[cfg(esp_idf_version_at_least_6_0_0)]
561                        mtu: 0,
562                    },
563                    None,
564                    false,
565                    None,
566                    None,
567                    None,
568                )
569            }
570        };
571
572        if let Some(ip_info) = ip_info.as_ref() {
573            esp_inherent_config.ip_info = ip_info;
574        }
575
576        // ESP-IDF copies the bridge configuration when creating the netif
577        #[cfg(all(esp_idf_esp_netif_tcpip_lwip, esp_idf_esp_netif_bridge_en))]
578        let mut bridge_info: bridgeif_config_t;
579        #[cfg(all(esp_idf_esp_netif_tcpip_lwip, esp_idf_esp_netif_bridge_en))]
580        if let NetifStack::Bridge(bridge_conf) = conf.stack {
581            bridge_info = bridgeif_config_t {
582                max_fdb_dyn_entries: bridge_conf.max_fdb_dyn_entries,
583                max_fdb_sta_entries: bridge_conf.max_fdb_sta_entries,
584                max_ports: bridge_conf.max_ports,
585            };
586
587            esp_inherent_config.flags |= esp_netif_flags_ESP_NETIF_FLAG_IS_BRIDGE;
588            esp_inherent_config.bridge_info = &mut bridge_info;
589        }
590
591        let cfg = esp_netif_config_t {
592            base: &esp_inherent_config,
593            driver: ptr::null(),
594            stack: conf.stack.default_raw_stack(),
595        };
596
597        let mut netif = Self {
598            handle: unsafe { esp_netif_new(&cfg).as_mut() }
599                .ok_or(EspError::from_infallible::<ESP_ERR_INVALID_ARG>())?,
600            _got_ip_event_id: conf.got_ip_event_id,
601            _lost_ip_event_id: conf.lost_ip_event_id,
602        };
603
604        if let Some(dns) = dns {
605            netif.set_dns(dns);
606
607            if dhcps {
608                // ESP-IDF has the DHCP server only with `CONFIG_LWIP_DHCPS`, which requires IPv4
609                #[cfg(esp_idf_lwip_dhcps)]
610                {
611                    #[cfg(esp_idf_version_major = "4")]
612                    let mut dhcps_dns_value: dhcps_offer_t = dhcps_offer_option_OFFER_DNS as _;
613
614                    // Strangely dhcps_offer_t and dhcps_offer_option_* are not included in ESP-IDF V5's bindings
615                    #[cfg(not(esp_idf_version_major = "4"))]
616                    let mut dhcps_dns_value: u8 = 2_u8;
617
618                    esp!(unsafe {
619                        esp_netif_dhcps_option(
620                            netif.handle,
621                            esp_netif_dhcp_option_mode_t_ESP_NETIF_OP_SET,
622                            esp_netif_dhcp_option_id_t_ESP_NETIF_DOMAIN_NAME_SERVER,
623                            &mut dhcps_dns_value as *mut _ as *mut _,
624                            core::mem::size_of_val(&dhcps_dns_value) as u32,
625                        )
626                    })?;
627                }
628            }
629        }
630
631        if let Some(secondary_dns) = secondary_dns {
632            netif.set_secondary_dns(secondary_dns);
633        }
634
635        if let Some(hostname) = hostname {
636            netif.set_hostname(hostname)?;
637        }
638
639        Ok(netif)
640    }

839/// A layer 2 bridge between netifs (its ports), e.g. between Ethernet and a Wi-Fi access point.
840///
841/// The bridge has a netif of its own, created with [`NetifStack::Bridge`] (see
842/// [`NetifConfiguration::bridge_default`]): it is this netif which gets the IP configuration of
843/// the device, while the netifs of the ports only pass frames to and from the bridge.
844///
845/// As in ESP-IDF's bridge example:
846/// - The netifs of the ports should have no IP configuration (`ip_configuration: None`), so
847///   that they run neither a DHCP client nor a DHCP server
848/// - The Ethernet ports should share their MAC address with the bridge netif, and should be in
849///   promiscuous mode (see [`EthDriver::set_promiscuous`](crate::eth::EthDriver::set_promiscuous))
850/// - Ports have to be added before they are started; the bridge starts with its first port
851///
852/// Requires `CONFIG_ESP_NETIF_BRIDGE_EN=y` and `CONFIG_LWIP_NUM_NETIF_CLIENT_DATA=1`.
853#[cfg(all(esp_idf_esp_netif_tcpip_lwip, esp_idf_esp_netif_bridge_en))]
854pub struct EspNetifBridge {
855    glue: esp_netif_br_glue_handle_t,
856    netif: EspNetif,
857}
858
859#[cfg(all(esp_idf_esp_netif_tcpip_lwip, esp_idf_esp_netif_bridge_en))]
860impl EspNetifBridge {
861    /// Create a bridge with the given bridge netif.
862    ///
863    /// Returns `ESP_ERR_INVALID_ARG` if the netif was not created with [`NetifStack::Bridge`].
864    pub fn new(netif: EspNetif) -> Result<Self, EspError> {
865        if unsafe { esp_netif_get_flags(netif.handle) } & esp_netif_flags_ESP_NETIF_FLAG_IS_BRIDGE
866            == 0
867        {
868            return Err(EspError::from_infallible::<ESP_ERR_INVALID_ARG>());
869        }
870
871        let glue = unsafe { esp_netif_br_glue_new() };
872        if glue.is_null() {
873            return Err(EspError::from_infallible::<ESP_ERR_NO_MEM>());
874        }
875
876        // The bridge netif has to be attached before any of the ports starts
877        if let Err(err) = esp!(unsafe { esp_netif_attach(netif.handle, glue as *mut _) }) {
878            unsafe { esp_netif_br_glue_del(glue) };
879
880            return Err(err);
881        }
882
883        Ok(Self { glue, netif })
884    }
885
886    /// Add a netif - e.g. the netif of an Ethernet driver - as a port of the bridge.
887    ///
888    /// # Safety
889    ///
890    /// The bridge keeps a pointer to the netif, so the netif must not be dropped before the
891    /// bridge.
892    pub unsafe fn add_port(&mut self, netif: &EspNetif) -> Result<(), EspError> {
893        esp!(esp_netif_br_glue_add_port(self.glue, netif.handle))
894    }
895
896    /// Add the netif of the Wi-Fi access point as a port of the bridge.
897    ///
898    /// A bridge can have only one Wi-Fi port, and it has to be the access point netif.
899    ///
900    /// # Safety
901    ///
902    /// The bridge keeps a pointer to the netif, so the netif must not be dropped before the
903    /// bridge.
904    pub unsafe fn add_wifi_port(&mut self, netif: &EspNetif) -> Result<(), EspError> {
905        esp!(esp_netif_br_glue_add_wifi_port(self.glue, netif.handle))
906    }
907
908    /// Return the bridge netif.
909    pub fn netif(&self) -> &EspNetif {
910        &self.netif
911    }
912
913    /// Return the bridge netif.
914    pub fn netif_mut(&mut self) -> &mut EspNetif {
915        &mut self.netif
916    }
917}

920impl Drop for EspNetifBridge {
921    fn drop(&mut self) {
922        // Stops the bridge netif, so it has to happen before the netif is destroyed
923        unsafe { esp_netif_br_glue_del(self.glue) };
924    }
925}

1104        let event_id = data.event_id as u32;
1105
1106        if event_id == ip_event_t_IP_EVENT_AP_STAIPASSIGNED {
1107            let event = unsafe {
1108                (data.payload.unwrap() as *const _ as *const ip_event_ap_staipassigned_t)
1109                    .as_ref()
1110                    .unwrap()
1111            };
1112

1114        } else if event_id == ip_event_t_IP_EVENT_STA_GOT_IP
1115            || event_id == ip_event_t_IP_EVENT_ETH_GOT_IP
1116            || event_id == ip_event_t_IP_EVENT_PPP_GOT_IP
1117        {

1119                (data.payload.unwrap() as *const _ as *const ip_event_got_ip_t)
1120                    .as_ref()
1121                    .unwrap()
1122            };
1123
1124            IpEvent::DhcpIpAssigned(DhcpIpAssignment(event))
1125        } else if event_id == ip_event_t_IP_EVENT_GOT_IP6 {
1126            let event = unsafe {
1127                (data.payload.unwrap() as *const _ as *const ip_event_got_ip6_t)
1128                    .as_ref()
1129                    .unwrap()
1130            };
1131
1132            IpEvent::DhcpIp6Assigned(DhcpIp6Assignment(event))
1133        } else if event_id == ip_event_t_IP_EVENT_PPP_LOST_IP {
1134            // ESP-IDF posts an `ip_event_got_ip_t` for PPP loss as well; only
1135            // its `esp_netif` member is populated.
1136            let event = unsafe {
1137                (data.payload.unwrap() as *const _ as *const ip_event_got_ip_t)
1138                    .as_ref()
1139                    .unwrap()
1140            };
1141
1142            IpEvent::DhcpIpDeassigned(event.esp_netif)
1143        } else if event_id == ip_event_t_IP_EVENT_STA_LOST_IP
1144            || event_id == ip_event_t_IP_EVENT_ETH_LOST_IP
1145        {
1146            let netif_handle_mut = unsafe {
1147                (data.payload.unwrap() as *const _ as *mut esp_netif_t)
1148                    .as_mut()
1149                    .unwrap()
1150            };
1151
1152            IpEvent::DhcpIpDeassigned(netif_handle_mut as *mut _)
1153        } else {
1154            ::log::warn!("IpEvent: unknown event ID {event_id}, ignoring");
1155            IpEvent::Other(event_id as i32)
1156        }
1157    }

1338    /// Bridges an [`EspNetifDriver`] to asynchronous byte streams.
1339    ///
1340    /// This adapter is intended for PPP byte streams. A modem driver must
1341    /// complete command negotiation and enter PPP data mode before calling
1342    /// [`Self::run`]. ESP-IDF's PPP input deframes the byte stream, so reads may
1343    /// contain partial frames, multiple frames, or splits within an escape
1344    /// sequence. Use a reasonably large receive buffer to avoid a heap
1345    /// allocation and TCP/IP mailbox message for every small read.
1346    ///
1347    /// Other netif types may require exactly one complete L2 frame per receive
1348    /// call and different buffer ownership. Do not use this adapter for them
1349    /// without checking their ESP-IDF input contract.
1350    ///
1351    /// ESP-NETIF invokes its transmit callback synchronously, while
1352    /// [`Write`] is asynchronous. Outbound packets are therefore
1353    /// copied into a bounded queue. If that queue is full, ESP-NETIF is notified
1354    /// with `ESP_ERR_NO_MEM` instead of blocking its TCP/IP task.
1355    ///
1356    /// A modem-specific driver remains responsible for entering PPP data mode. For
1357    /// example, after obtaining an `a76xx::PppIo` stream from `connect_ppp`:
1358    ///
1359    /// ```ignore
1360    /// let mut bridge = AsyncEspNetifChannel::<_, 8>::new(
1361    ///     EspNetif::new(NetifStack::Ppp)?,
1362    ///     |netif| netif.set_ppp_conf(&PppConfiguration::default()),
1363    /// )?;
1364    /// bridge.driver_mut().start()?;
1365    ///
1366    /// let mut rx_buffer = [0; 1536];
1367    /// bridge.run(&mut ppp_io, &mut rx_buffer).await?;
1368    /// ```
1369    pub struct AsyncEspNetifChannel<'d, T, const TX_QUEUE_SIZE: usize = ASYNC_NETIF_TX_QUEUE_SIZE>
1370    where
1371        T: BorrowMut<EspNetif>,
1372    {
1373        driver: EspNetifDriver<'d, T>,

1390    impl<'d, T, const TX_QUEUE_SIZE: usize> AsyncEspNetifChannel<'d, T, TX_QUEUE_SIZE>
1391    where
1392        T: BorrowMut<EspNetif>,
1393    {
1394        /// Creates an asynchronous adapter which may borrow non-static data from
1395        /// `post_attach_cfg`.
1396        ///
1397        /// # Safety
1398        ///
1399        /// This has the same leak-safety requirements as
1400        /// [`EspNetifDriver::new_nonstatic`]. The adapter must not be forgotten.
1401        pub fn new_nonstatic<P>(netif: T, post_attach_cfg: P) -> Result<Self, EspError>
1402        where
1403            P: FnMut(&mut EspNetif) -> Result<(), EspError> + Send + 'd,
1404        {
1405            if TX_QUEUE_SIZE == 0 {
1406                return Err(EspError::from_infallible::<ESP_ERR_INVALID_ARG>());
1407            }
1408
1409            let tx_queue = Arc::new(Channel::new());
1410            let callback_queue = tx_queue.clone();
1411            let driver = EspNetifDriver::new_nonstatic(netif, post_attach_cfg, move |data| {
1412                callback_queue
1413                    .try_send(Vec::from(data))
1414                    .map_err(|_| EspError::from_infallible::<ESP_ERR_NO_MEM>())
1415            })?;
1416
1417            Ok(Self { driver, tx_queue })
1418        }
1419
1420        /// Runs both directions of the bridge over one bidirectional PPP stream.
1421        ///
1422        /// This is convenient for transports such as `a76xx::PppIo` which implement
1423        /// both [`embedded_io_async::Read`] and [`embedded_io_async::Write`] but do
1424        /// not expose separate halves. The transport's read operation must be
1425        /// cancellation-safe: it is cancelled whenever ESP-NETIF has an outbound
1426        /// packet ready to write.
1427        ///
1428        /// The method returns successfully if the transport reports end-of-stream
1429        /// by returning zero bytes.
1430        pub async fn run<IO>(
1431            &self,
1432            mut io: IO,
1433            rx_buffer: &mut [u8],
1434        ) -> Result<(), AsyncEspNetifChannelError<IO::Error>>
1435        where
1436            IO: Read + Write,
1437        {
1438            loop {
1439                match select(io.read(rx_buffer), self.tx_queue.receive()).await {
1440                    Either::First(result) => {
1441                        let len = result.map_err(AsyncEspNetifChannelError::Read)?;
1442                        if len == 0 {
1443                            return Ok(());
1444                        }
1445                        self.driver
1446                            .rx(&rx_buffer[..len])
1447                            .map_err(AsyncEspNetifChannelError::Netif)?;
1448                    }
1449                    Either::Second(packet) => io
1450                        .write_all(&packet)
1451                        .await
1452                        .map_err(AsyncEspNetifChannelError::Write)?,
1453                }
1454            }
1455        }

1460        }

1465        }

1542        ///
1543        /// Arguments:
1544        /// - `netif` is the `EspNetif` instance that the driver will manage
1545        /// - `got_ip_event_id` and `lost_ip_event_id` are the event IDs that the driver
1546        ///   will listen to so that it can connect/disconnect the netif upon receival
1547        ///   / loss of IP

1549        ///   after the netif is attached. For example, for a PPP netif, the post attach
1550        ///   configuration might want to invoke `EspNetif::set_ppp_conf`.
1551        /// - `tx` is the callback that the driver will call when it wants to ingest a packet
1552        ///   into the underlying transport
1553        ///

1555        /// ```ignore
1556        /// let (uart_rx, uart_tx) = uart.into_split();

1567        /// loop {
1568        ///     let mut buffer = [0; 128];
1569        ///     let len = uart_rx.read(&mut buffer)?;
1570        ///     driver.rx(&buffer[..len])?;
1571        /// }
1572        /// ```
1573        ///
1574        /// # Safety
1575        ///
1576        /// This method - in contrast to method `new` - allows the user to pass
1577        /// non-static callbacks/closures. This enables users to borrow
1578        /// - in the closure - variables that live on the stack - or more generally - in the same
1579        ///   scope where the service is created.
1580        ///
1581        /// HOWEVER: care should be taken NOT to call `core::mem::forget()` on the service,
1582        /// as that would immediately lead to an UB (crash).
1583        /// Also note that forgetting the service might happen with `Rc` and `Arc`
1584        /// when circular references are introduced: <https://github.com/rust-lang/rust/issues/24456>
1585        ///
1586        /// The reason is that the closure is actually sent to a hidden ESP IDF thread.
1587        /// This means that if the service is forgotten, Rust is free to e.g. unwind the stack
1588        /// and the closure now owned by this other thread will end up with references to variables that no longer exist.
1589        ///
1590        /// The destructor of the service takes care - prior to the service being dropped and e.g.
1591        /// the stack being unwind - to remove the closure from the hidden thread and destroy it.
1592        /// Unfortunately, when the service is forgotten, the un-subscription does not happen
1593        /// and invalid references are left dangling.
1594        ///
1595        /// This "local borrowing" will only be possible to express in a safe way once/if `!Leak` types
1596        /// are introduced to Rust (i.e. the impossibility to "forget" a type and thus not call its destructor).
1597        pub fn new_nonstatic<P, F>(netif: T, post_attach_cfg: P, tx: F) -> Result<Self, EspError>
1598        where
1599            P: FnMut(&mut EspNetif) -> Result<(), EspError> + Send + 'd,
1600            F: FnMut(&[u8]) -> Result<(), EspError> + Send + 'd,
1601        {
1602            let mut inner = alloc::boxed::Box::new(EspNetifDriverInner {
1603                base: esp_netif_driver_base_t {
1604                    netif: netif.borrow().handle(),
1605                    post_attach: Some(EspNetifDriverInner::<T>::raw_post_attach),
1606                },
1607                netif,
1608                post_attach_cfg: alloc::boxed::Box::new(post_attach_cfg),
1609                tx: alloc::boxed::Box::new(tx),
1610            });
1611
1612            let inner_ptr = inner.as_mut() as *mut _ as *mut core::ffi::c_void;
1613
1614            if let Some(got_ip_event_id) = inner.netif.borrow()._got_ip_event_id {
1615                esp!(unsafe {
1616                    esp_event_handler_register(
1617                        IP_EVENT,
1618                        got_ip_event_id.get() as _,
1619                        Some(esp_netif_action_connected),
1620                        inner.netif.borrow().handle() as *mut core::ffi::c_void,
1621                    )
1622                })?;
1623            }
1624
1625            if let Some(lost_ip_event_id) = inner.netif.borrow()._lost_ip_event_id {
1626                esp!(unsafe {
1627                    esp_event_handler_register(
1628                        IP_EVENT,
1629                        lost_ip_event_id.get() as _,

1631                        inner.netif.borrow().handle() as *mut core::ffi::c_void,
1632                    )
1633                })?;
1634            }
1635
1636            esp!(unsafe { esp_netif_attach(inner.netif.borrow().handle(), inner_ptr) })?;
1637
1638            Ok(Self {
1639                inner,
1640                started: false,
1641            })
1642        }

1645        ///
1646        /// The packet can arrive from anywhere, but with say - a PPP netif -
1647        /// it would be a PPP packet arriving typically from UART, by reading from it.

1782        unsafe extern "C" fn raw_tx(
1783            h: *mut core::ffi::c_void,
1784            buffer: *mut core::ffi::c_void,
1785            len: usize,
1786        ) -> i32 {
1787            let this = h as *mut Self;
1788            let data = core::slice::from_raw_parts(buffer as *mut u8, len);
1789            // Only the callback field is accessed here. Creating `&mut Self`
1790            // would incorrectly claim exclusive access to the netif while its
1791            // RX side can be active on another task.
1792            let tx = unsafe { &mut *core::ptr::addr_of_mut!((*this).tx) };
1793            let result = match tx(data) {
1794                Ok(_) => ESP_OK,
1795                Err(e) => e.code(),
1796            };
1797
1798            // TODO: Might not be necessary, but if I remember correctly, the Netif API
1799            // wanted that _we_ free the buffer; in any case needs to be compared with the C ESP Modem code
1800            // free(buffer);
1801
1802            result
1803        }

1808        ) -> i32 {

1811                Ok(_) => ESP_OK,
1812                Err(e) => e.code(),
