80        // TODO: replace with `CStr::from_bytes_until_nul` when stabilized
81        let mut nul_pos = 0;
82        while nul_pos < bytes.len() {
83            if bytes[nul_pos] == 0 {
84                // TODO: replace with `<[u8]>::split_at(nul_pos + 1)` when const stabilized

87            }
88            nul_pos += 1;
89        }
90        panic!("PEM certificates should end with a NIL (`\\0`) ASCII character.")
91    }

132    /// A private key for TLS client authentication which never leaves the hardware.
133    ///
134    /// It is used instead of a private key in memory, so it is mutually exclusive with the
135    /// private key setting of the configuration it is used with. The client certificate

138    /// Which variants are available depends on the chip and on the ESP-IDF configuration.

140    pub enum HwPrivateKey {
141        /// An RSA key used through the Digital Signature (DS) peripheral
142        #[cfg(all(esp_idf_esp_tls_use_ds_peripheral, esp_idf_version_at_least_5_1_0))]
143        Ds(EspDsKey),
144        /// An ECDSA key used through the ECDSA peripheral
145        /// (requires `CONFIG_MBEDTLS_HARDWARE_ECDSA_SIGN=y`)
146        #[cfg(esp_idf_mbedtls_hardware_ecdsa_sign)]
147        Ecdsa(EspEcdsaKey),
148    }

150    /// An RSA private key, encrypted for the Digital Signature (DS) peripheral of this chip.
151    ///
152    /// The key is provisioned once, e.g. with `configure_esp_secure_cert.py` of
153    /// `esp_secure_cert_mgr`: an HMAC key is burned into an eFuse key block, and the RSA key
154    /// is encrypted with a key derived from it. The DS peripheral decrypts and uses the RSA key
155    /// internally, and the encrypted key is useless on any other chip.
156    ///
157    /// Clones share the encrypted key, so cloning is cheap.
158    #[cfg(all(esp_idf_esp_tls_use_ds_peripheral, esp_idf_version_at_least_5_1_0))]
159    #[derive(Clone)]
160    pub struct EspDsKey {
161        data: Arc<esp_ds_data_t>,
162        hmac_key_id: u8,
163        rsa_key_bits: u16,
164    }
165
166    #[cfg(all(esp_idf_esp_tls_use_ds_peripheral, esp_idf_version_at_least_5_1_0))]
167    impl EspDsKey {
168        /// Create a key from the outputs of its provisioning.
169        ///
170        /// - `ciphertext`: the encrypted key parameters
171        /// - `iv`: the initialization vector they were encrypted with
172        /// - `rsa_key_bits`: the length of the RSA key in bits (1024, 2048, 3072 or 4096)
173        /// - `hmac_key_id`: the index (0 to 5) of the eFuse key block (`KEY0` to `KEY5`) holding
174        ///   the HMAC key
175        ///
176        /// Returns `ESP_ERR_INVALID_ARG` if any of these does not match the DS peripheral.
177        pub fn new(
178            ciphertext: &[u8],
179            iv: &[u8],
180            rsa_key_bits: u16,
181            hmac_key_id: u8,
182        ) -> Result<Self, EspError> {
183            let rsa_length = match rsa_key_bits {
184                1024 => esp_digital_signature_length_t_ESP_DS_RSA_1024,
185                2048 => esp_digital_signature_length_t_ESP_DS_RSA_2048,
186                3072 => esp_digital_signature_length_t_ESP_DS_RSA_3072,
187                4096 => esp_digital_signature_length_t_ESP_DS_RSA_4096,
188                _ => return Err(EspError::from_infallible::<ESP_ERR_INVALID_ARG>()),
189            };
190
191            let mut data = Arc::new(esp_ds_data_t::default());
192            let raw = Arc::get_mut(&mut data).unwrap();
193
194            if ciphertext.len() != raw.c.len()
195                || iv.len() != core::mem::size_of_val(&raw.iv)
196                || hmac_key_id as u32 >= hmac_key_id_t_HMAC_KEY_MAX
197            {
198                return Err(EspError::from_infallible::<ESP_ERR_INVALID_ARG>());
199            }
200
201            raw.rsa_length = rsa_length;
202            raw.c.copy_from_slice(ciphertext);
203
204            // The IV is stored as native-endian words; the chips are little-endian
205            for (word, bytes) in raw.iv.iter_mut().zip(iv.chunks_exact(4)) {
206                *word = u32::from_le_bytes(bytes.try_into().unwrap());
207            }
208
209            Ok(Self {
210                data,
211                hmac_key_id,
212                rsa_key_bits,
213            })
214        }
215
216        /// Return the length of the RSA key in bits.
217        pub const fn rsa_key_bits(&self) -> u16 {
218            self.rsa_key_bits
219        }
220
221        /// Return the index of the eFuse key block holding the HMAC key.
222        pub const fn hmac_key_id(&self) -> u8 {
223            self.hmac_key_id
224        }
225    }

270    /// The C-side state of a [`HwPrivateKey`].
271    ///
272    /// ESP-IDF keeps a pointer to it in the configuration of the client using the key, so it has
273    /// to live as long as that client.
274    #[derive(Default)]
275    pub(crate) struct RawHwPrivateKey {
276        #[cfg(all(esp_idf_esp_tls_use_ds_peripheral, esp_idf_version_at_least_5_1_0))]
277        ds: Option<(EspDsKey, Box<DsDataCtx>)>,
278        #[cfg(esp_idf_mbedtls_hardware_ecdsa_sign)]
279        ecdsa_efuse_block: Option<u8>,
280    }

282    impl RawHwPrivateKey {
283        #[allow(unused_variables)]
284        pub(crate) fn new(key: Option<&HwPrivateKey>) -> Self {
285            Self {
286                #[cfg(all(esp_idf_esp_tls_use_ds_peripheral, esp_idf_version_at_least_5_1_0))]
287                ds: match key {
288                    Some(HwPrivateKey::Ds(key)) => {
289                        let ctx = Box::new(DsDataCtx {
290                            esp_ds_data: Arc::as_ptr(&key.data) as *mut _,
291                            efuse_key_id: key.hmac_key_id,
292                            rsa_length_bits: key.rsa_key_bits,
293                        });
294
295                        Some((key.clone(), ctx))
296                    }
297                    #[allow(unreachable_patterns)]
298                    _ => None,
299                },
300                #[cfg(esp_idf_mbedtls_hardware_ecdsa_sign)]
301                ecdsa_efuse_block: match key {
302                    Some(HwPrivateKey::Ecdsa(key)) => Some(key.efuse_block),
303                    #[allow(unreachable_patterns)]
304                    _ => None,
305                },
306            }
307        }
308
309        /// The `ds_data` to set in the ESP-IDF configuration; null when not using the DS
310        /// peripheral.
311        #[cfg(all(esp_idf_esp_tls_use_ds_peripheral, esp_idf_version_at_least_5_1_0))]
312        pub(crate) fn ds_data(&self) -> *mut core::ffi::c_void {
313            match &self.ds {
314                Some((_, ctx)) => &**ctx as *const DsDataCtx as *mut _,
315                None => core::ptr::null_mut(),
316            }
317        }
318
319        /// The eFuse block to set in the ESP-IDF configuration when using the ECDSA peripheral.
320        #[cfg(esp_idf_mbedtls_hardware_ecdsa_sign)]
321        pub(crate) fn ecdsa_efuse_block(&self) -> Option<u8> {
322            self.ecdsa_efuse_block
323        }
324    }

388        pub const fn new() -> Self {
389            Self {

392                client_cert: None,
393                client_key: None,

401                keep_alive_cfg: None,
402                psk_hint_key: None,
403                #[cfg(esp_idf_mbedtls_certificate_bundle)]
404                use_crt_bundle_attach: true,
405                #[cfg(feature = "alloc")]
406                hw_private_key: None,
407                is_plain_tcp: false,
408            }
409        }
410
411        fn try_into_raw(
412            &self,
413            bufs: &mut RawConfigBufs,
414            alpn: &mut SessionAlpnProtos,
415        ) -> Result<sys::esp_tls_cfg, EspError> {
416            let mut rcfg: sys::esp_tls_cfg = Default::default();
417
418            if let Some(ca_cert) = self.ca_cert {
419                rcfg.__bindgen_anon_1.cacert_buf = ca_cert.data().as_ptr();
420                rcfg.__bindgen_anon_2.cacert_bytes = ca_cert.data().len() as u32;
421            }
422
423            if let Some(client_cert) = self.client_cert {
424                rcfg.__bindgen_anon_3.clientcert_buf = client_cert.data().as_ptr();
425                rcfg.__bindgen_anon_4.clientcert_bytes = client_cert.data().len() as u32;
426            }
427
428            if let Some(client_key) = self.client_key {
429                rcfg.__bindgen_anon_5.clientkey_buf = client_key.data().as_ptr();
430                rcfg.__bindgen_anon_6.clientkey_bytes = client_key.data().len() as u32;
431            }
432
433            if let Some(ckp) = self.client_key_password {
434                rcfg.clientkey_password = ckp.as_ptr();
435                rcfg.clientkey_password_len = ckp.len() as u32;
436            }
437
438            #[cfg(feature = "alloc")]
439            {
440                if self.hw_private_key.is_some() && self.client_key.is_some() {
441                    return Err(EspError::from_infallible::<{ sys::ESP_ERR_INVALID_ARG }>());
442                }
443
444                bufs.hw_private_key = super::RawHwPrivateKey::new(self.hw_private_key.as_ref());
445
446                #[cfg(all(esp_idf_esp_tls_use_ds_peripheral, esp_idf_version_at_least_5_1_0))]
447                {
448                    rcfg.ds_data = bufs.hw_private_key.ds_data();
449                }
450
451                #[cfg(esp_idf_mbedtls_hardware_ecdsa_sign)]
452                if let Some(efuse_block) = bufs.hw_private_key.ecdsa_efuse_block() {
453                    rcfg.use_ecdsa_peripheral = true;
454                    rcfg.ecdsa_key_efuse_blk = efuse_block;
455                }
456            }
457
458            if let Some(protos) = self.alpn_protos {
459                rcfg.alpn_protos = alpn.get_or_init(protos)?;
460            }
461
462            rcfg.non_block = self.non_block;
463            rcfg.use_secure_element = self.use_secure_element;
464            rcfg.timeout_ms = self.timeout_ms as i32;
465            rcfg.use_global_ca_store = self.use_global_ca_store;
466
467            if let Some(common_name) = self.common_name {
468                rcfg.common_name =
469                    cstr_from_str_truncating(common_name, &mut bufs.common_name_buf).as_ptr();
470            }
471
472            rcfg.skip_common_name = self.skip_common_name;
473
474            if let Some(kac) = &self.keep_alive_cfg {
475                bufs.keep_alive_cfg = sys::tls_keep_alive_cfg {
476                    keep_alive_enable: kac.enable,
477                    keep_alive_idle: kac.idle.as_secs() as i32,
478                    keep_alive_interval: kac.interval.as_secs() as i32,
479                    keep_alive_count: kac.count as i32,
480                };
481                rcfg.keep_alive_cfg = &mut bufs.keep_alive_cfg as *mut _;
482            }
483
484            #[cfg(any(
485                esp_idf_esp_tls_psk_verification,
486                esp_idf_version_major = "4",
487                esp_idf_version = "5.0",
488                esp_idf_version = "5.1",
489                esp_idf_version = "5.2",
490                esp_idf_version = "5.3",
491                esp_idf_version = "5.4",
492            ))]
493            if let Some(psk) = &self.psk_hint_key {
494                bufs.psk_hint_key = sys::psk_key_hint {
495                    key: psk.key.as_ptr(),
496                    key_size: psk.key.len(),
497                    hint: psk.hint.as_ptr(),
498                };
499                rcfg.psk_hint_key = &bufs.psk_hint_key as *const _;
500            }
501
502            #[cfg(esp_idf_mbedtls_certificate_bundle)]
503            if self.use_crt_bundle_attach {
504                rcfg.crt_bundle_attach = Some(sys::esp_crt_bundle_attach);
505            }
506
507            rcfg.is_plain_tcp = self.is_plain_tcp;
508
509            #[cfg(esp_idf_comp_lwip_enabled)]
510            {
511                rcfg.if_name = core::ptr::null_mut();
512            }
513
514            Ok(rcfg)
515        }
516    }

566    ///
567    /// mbedTLS reads the list during the handshake, and points into it until the session is
568    /// destroyed. The list is therefore on the heap, where moving the `EspTls` does not move it.

615        pub fn alpn_proto(&self) -> Option<&str> {
616            let p = CStr::from_bytes_until_nul(self.alpn.as_slice()).unwrap();
617            // Safety: the bytes always come from a user supplied &str.

647            esp_idf_comp_esp_tls_enabled,

649            esp_idf_mbedtls_ssl_alpn
650        ))]

652        unsafe fn extract_alpn(raw: *mut sys::esp_tls) -> Option<AlpnBuf> {
653            let raw: *mut sys::mbedtls_ssl_context =
654                unsafe { sys::esp_tls_get_ssl_context(raw) }.cast();
655
656            if raw.is_null() {
657                return None;
658            }
659
660            let chosen = unsafe { sys::mbedtls_ssl_get_alpn_protocol(raw) };
661            if chosen.is_null() {
662                return None;
663            }
664
665            let mut proto = AlpnBuf::default();
666            let chosen = unsafe { CStr::from_ptr(chosen) };
667            let chosen_bytes = chosen.to_bytes_with_nul();
668            if chosen_bytes.len() > proto.len() {
669                return None;
670            }
671
672            proto[..chosen_bytes.len()].copy_from_slice(chosen_bytes);
673
674            Some(proto)
675        }

1249        /// Create a new `AsyncEspTls` instance adopting the supplied socket.
1250        /// The socket should be in a connected state.
1251        ///
1252        /// # Errors
1253        ///
1254        /// * `ESP_ERR_NO_MEM` if not enough memory to create the TLS connection
1255        pub fn adopt(socket: S) -> Result<Self, EspError> {
1256            Ok(Self(crate::private::mutex::Mutex::new(EspTls::adopt(
1257                socket,
1258            )?)))

1261        /// Establish a TLS/SSL connection using the adopted socket.
1262        ///
1263        /// # Errors
1264        ///
1265        /// * `ESP_ERR_INVALID_SIZE` if `cfg.alpn_protos` exceeds 9 elements or avg 10 bytes/ALPN
1266        /// * `ESP_FAIL` if connection could not be established
1267        pub async fn negotiate(
1268            &mut self,
1269            hostname: &str,
1270            cfg: &Config<'_>,
1271        ) -> Result<CompletedHandshake, EspError> {
1272            struct AssertSend<T>(T);
1273            unsafe impl<T> Send for AssertSend<T> {}
1274
1275            let mut bufs = RawConfigBufs::default();
1276            let mut rcfg: AssertSend<sys::esp_tls_cfg> =
1277                AssertSend(cfg.try_into_raw(&mut bufs, &mut self.0.get_mut().alpn_protos)?);
1278
1279            // It is a bit unintuitive, but when an async socket is being adopted, `non_block` should be set to false.
1280            //
1281            // Background:
1282            // `non_block = true` is only used at one place in the ESP IDF code and that is to run
1283            // a check - with `select` - whether the socket is really connected.
1284            // However, we want to avoid the `select()` call, as
1285            // (a) It won't work, because we jump directly into the ESP_TLS_CONNECTING state as we adopt a socket.
1286            //.    As a side effect, the select() call is not properly initialized.
1287            // (b) The adopted socket might be registered in a select() loop already.
1288            //
1289            // Avoiding the connectivity check with `select()` should be fine, as the adopted socket
1290            // must be already connected anyway (API requirement).
1291            rcfg.0.non_block = false;
1292
1293            let res = loop {
1294                let res = self
1295                    .0
1296                    .get_mut()
1297                    .internal_connect(hostname, 0, true, &rcfg.0);
1298
1299                match res {
1300                    Err(e) => self.wait(e).await?,
1301                    other => break other,
1302                }
1303            };
1304
1305            // Make sure buffers are held long enough
1306            #[allow(clippy::drop_non_drop)]
1307            drop(bufs);
1308
1309            res
1310        }

1312        /// Establish a TLS/SSL connection using the adopted socket, acting as the server.
1313        ///
1314        /// Requires ESP-IDF >= 5.5.0 with mbedTLS.
1315        ///
1316        /// Note that the handshake is not bounded in time (`cfg.tls_handshake_timeout_ms`
1317        /// is only honored by the blocking `EspTls::negotiate_server`), so a
1318        /// misbehaving peer can keep the negotiation going indefinitely.
1319        /// Callers which cannot afford that should race this future against a timer.
1320        ///
1321        /// # Errors
1322        ///
1323        /// * `ESP_FAIL` if the connection could not be established
1324        #[cfg(all(esp_idf_version_at_least_5_5_0, esp_idf_esp_tls_using_mbedtls))]
1325        pub async fn negotiate_server(&mut self, cfg: &ServerConfig<'_>) -> Result<(), EspError> {
1326            self.0.get_mut().negotiate_server_init(cfg)?;
1327
1328            loop {
1329                let res = self.0.get_mut().negotiate_server_continue();
1330
1331                match res {
1332                    Err(e) => self.wait(e).await?,
1333                    Ok(()) => break Ok(()),
1334                }
1335            }
1336        }

1392        fn poll_wait(&self, ctx: &mut Context<'_>, error: EspError) -> Poll<Result<(), EspError>> {
1393            const EWOULDBLOCK_I32: i32 = EWOULDBLOCK as i32;
1394
1395            match error.code() {
1396                // EWOULDBLOCK models the "0" return code of esp_mbedtls_handshake() which does not allow us
1397                // to figure out whether we need the socket to become readable or writable
1398                // The code below is therefore a hack which just waits with a timeout for the socket to (eventually)
1399                // become readable as we actually don't even know if that's what esp_tls wants
1400                EWOULDBLOCK_I32 => {
1401                    let res = self.0.lock().socket.poll_writable(ctx);
1402                    crate::hal::delay::FreeRtos::delay_ms(0);
1403                    res
1404                }
1405                ESP_TLS_ERR_SSL_WANT_READ => self.0.lock().socket.poll_readable(ctx),
1406                ESP_TLS_ERR_SSL_WANT_WRITE => self.0.lock().socket.poll_writable(ctx),
1407                _ => Poll::Ready(Err(error)),
1408            }
1409        }
