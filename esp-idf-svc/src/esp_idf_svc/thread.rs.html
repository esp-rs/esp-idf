269/// This struct provides a safe wrapper over the ESP IDF Thread C driver.
270///
271/// The driver works on Layer 2 (Data Link) in the OSI model, in that it provides
272/// facilities for sending and receiving ethernet packets over the Thread radio.
273///
274/// For most use cases, utilizing `EspThread` - which provides a networking (IP)
275/// layer as well - should be preferred. Using `ThreadDriver` directly is beneficial
276/// only when one would like to utilize a custom, non-STD network stack like `smoltcp`.
277///
278/// The driver can work in two modes:
279/// - RCP (Radio Co-Processor) mode: The driver operates as a co-processor to the host,
280///   which is expected to be another chip connected to ours via SPI or UART. This is
281///   of course only supported with MCUs that do have a Thread radio, like the esp32c6, esp32h2, esp32h21, esp32h4 and esp32s31
282/// - Host mode: The driver operates as a host, and if the chip does not have a Thread radio
283///   it has to be connected via SPI or USB to a chip which runs the Thread stack in RCP mode
284pub struct ThreadDriver<'d, T>
285where
286    T: Mode,
287{
288    inner: UnsafeCell<Box<ThreadDriverInner>>,
289    //_subscription: EspSubscription<'static, System>,
290    _nvs: EspDefaultNvsPartition,
291    _mounted_event_fs: Arc<MountedEventfs>,
292    _mode: T,
293    _p: PhantomData<&'d mut ()>,
294}

535    /// Perform an active scan for Thread networks
536    ///
537    /// The callback will be called for each found network
538    /// At the end of the scan, the callback will be called with `None`
539    pub fn scan<F: FnMut(Option<ActiveScanResult>) + Send + 'static>(
540        &self,
541        callback: F,
542    ) -> Result<(), EspError> {
543        let mut inner = self.inner();
544
545        if inner.scan_cb.is_some() {
546            return Err(EspError::from_infallible::<ESP_ERR_INVALID_STATE>());
547        }
548
549        #[allow(clippy::type_complexity)]
550        let callback: Box<Box<dyn FnMut(Option<ActiveScanResult>) + Send + 'static>> =
551            Box::new(Box::new(callback));
552
553        // Store callback where on_active_scan_result can get to it
554        inner.scan_cb = Some(callback);
555
556        match ot_esp!(unsafe {
557            otLinkActiveScan(
558                esp_openthread_get_instance(),
559                0xffff_ffffu32, // All channels
560                200,            // ms scan per channel
561                Some(Self::on_active_scan_result),
562                &mut *inner as *mut ThreadDriverInner as *mut c_void,
563            )
564        }) {
565            Ok(()) => Ok(()),
566            Err(err) => {
567                // Clean up inner if we fail to start the scan
568                inner.scan_cb = None;
569                Err(err)
570            }
571        }
572    }

627    /// Send an Ipv6 raw packet over Thread
628    pub fn tx(&self, packet: &[u8]) -> Result<(), EspError> {
629        let _lock = self.inner();
630
631        let message =
632            unsafe { otIp6NewMessage(esp_openthread_get_instance(), core::ptr::null_mut()) };
633        if message.is_null() {
634            Err(EspError::from_infallible::<ESP_FAIL>())?;
635        }
636
637        let result = ot_esp!(unsafe {
638            otMessageAppend(message, packet.as_ptr() as *const _, packet.len() as _)
639        })
640        .and_then(|_| ot_esp!(unsafe { otIp6Send(esp_openthread_get_instance(), message) }));
641
642        unsafe { otMessageFree(message) };
643
644        result
645    }

661    /// This method - in contrast to method `set_rx_callback` - allows the user to pass
662    /// non-static callback/closure. This enables users to borrow
663    /// - in the closure - variables that live on the stack - or more generally - in the same
664    ///   scope where the service is created.
665    ///
666    /// HOWEVER: care should be taken NOT to call `core::mem::forget()` on the service,
667    /// as that would immediately lead to an UB (crash).
668    /// Also note that forgetting the service might happen with `Rc` and `Arc`
669    /// when circular references are introduced: <https://github.com/rust-lang/rust/issues/24456>
670    ///
671    /// The reason is that the closure is actually sent to a hidden ESP IDF thread.
672    /// This means that if the service is forgotten, Rust is free to e.g. unwind the stack
673    /// and the closure now owned by this other thread will end up with references to variables that no longer exist.
674    ///
675    /// The destructor of the service takes care - prior to the service being dropped and e.g.
676    /// the stack being unwind - to remove the closure from the hidden thread and destroy it.
677    /// Unfortunately, when the service is forgotten, the un-subscription does not happen
678    /// and invalid references are left dangling.
679    ///
680    /// This "local borrowing" will only be possible to express in a safe way once/if `!Leak` types
681    /// are introduced to Rust (i.e. the impossibility to "forget" a type and thus not call its destructor).
682    pub fn set_nonstatic_rx_callback<R>(&self, callback: Option<R>) -> Result<(), EspError>
683    where
684        R: FnMut(Ipv6Incoming) + Send + 'd,
685    {
686        let mut inner = self.inner();
687
688        Self::internal_set_rx_callback(&mut inner, callback)
689    }
690
691    // NOTE: Methods starting with `internal_` have to be called only when the OpenThread lock is held
692
693    fn internal_set_rx_callback<R>(
694        inner: &mut ThreadDriverInner,
695        callback: Option<R>,
696    ) -> Result<(), EspError>
697    where
698        R: FnMut(Ipv6Incoming) + Send + 'd,
699    {
700        if let Some(callback) = callback {
701            #[allow(clippy::type_complexity)]
702            let callback: Box<Box<dyn FnMut(Ipv6Incoming) + Send + 'd>> =
703                Box::new(Box::new(callback));
704
705            #[allow(clippy::type_complexity)]
706            let callback: Box<Box<dyn FnMut(Ipv6Incoming) + Send + 'static>> =
707                unsafe { core::mem::transmute(callback) };
708
709            // Stick the callback where Self::on_address/Self::on_packet can get to it
710            inner.ipv6_cb = Some(callback);
711
712            unsafe {
713                otIp6SetAddressCallback(
714                    esp_openthread_get_instance(),
715                    Some(Self::on_address),
716                    &mut *inner as *mut ThreadDriverInner as *mut c_void,
717                );
718                otIp6SetReceiveCallback(
719                    esp_openthread_get_instance(),
720                    Some(Self::on_packet),
721                    &mut *inner as *mut ThreadDriverInner as *mut c_void,
722                );
723                otIp6SetReceiveFilterEnabled(esp_openthread_get_instance(), true);
724
725                // TODO otIcmp6SetEchoMode(esp_openthread_get_instance(), OT_ICMP6_ECHO_HANDLER_RLOC_ALOC_ONLY);
726            }
727        } else {
728            unsafe {
729                otIp6SetAddressCallback(esp_openthread_get_instance(), None, core::ptr::null_mut());
730                otIp6SetReceiveCallback(esp_openthread_get_instance(), None, core::ptr::null_mut());
731                otIp6SetReceiveFilterEnabled(esp_openthread_get_instance(), true);
732                // TODO otIcmp6SetEchoMode(esp_openthread_get_instance(), OT_ICMP6_ECHO_HANDLER_RLOC_ALOC_ONLY);
733            }
734
735            inner.ipv6_cb = None;
736        }
737
738        Ok(())
739    }

742        inner: &mut ThreadDriverInner,
743        active: bool,
744        buf: &mut [u8],
745    ) -> Result<usize, EspError> {
746        let dataset_buf = &mut inner.dataset_buf;
747
748        ot_esp!(unsafe {
749            if active {
750                otDatasetGetActiveTlvs(esp_openthread_get_instance(), dataset_buf)
751            } else {
752                otDatasetGetPendingTlvs(esp_openthread_get_instance(), dataset_buf)
753            }
754        })?;
755
756        let len = dataset_buf.mLength as usize;
757        if buf.len() < len {
758            Err(EspError::from_infallible::<ESP_ERR_INVALID_ARG>())?;
759        }
760
761        buf[..len].copy_from_slice(&dataset_buf.mTlvs[..len]);
762
763        Ok(len)
764    }

792    /// Populates the internal OT TLV dataset structure with the given dataset in HEX-TLV str format.
793    fn fill_dataset_tlv_hexstr(
794        dataset_buf: &mut otOperationalDatasetTlvs,
795        dataset: &str,
796    ) -> Result<(), EspError> {
797        let dataset = dataset.trim();
798        let mut offset = 0;
799
800        for (chf, chs) in dataset
801            .chars()
802            .step_by(2)
803            .zip(dataset.chars().skip(1).step_by(2))
804        {
805            let byte = (chf
806                .to_digit(16)
807                .ok_or(ot_esp_err(otError_OT_ERROR_INVALID_ARGS))?
808                << 4)
809                | chs
810                    .to_digit(16)
811                    .ok_or(ot_esp_err(otError_OT_ERROR_INVALID_ARGS))?;
812
813            if offset >= dataset_buf.mTlvs.len() {
814                Err(ot_esp_err(otError_OT_ERROR_NO_BUFS))?;
815            }
816
817            dataset_buf.mTlvs[offset] = byte as _;
818            offset += 1;
819        }
820
821        dataset_buf.mLength = offset as _;
822
823        Ok(())
824    }

841            } else {
842                ipv6_cb(Ipv6Incoming::AddressRemoved(address));
843            }
844        }
845    }

1258        {

1263            }

1282

1299        {
1300            let inner = self.inner();
1301
1302            if !*inner.started.lock() {
1303                Err(EspError::from_infallible::<ESP_ERR_INVALID_STATE>())?;
1304            }
1305

1307            #[allow(unused_assignments)]

1319                    esp_openthread_mainloop_exit();
1320                }
1321
1322                stop_supported = true;
1323            }

1332
1333            let started = inner.started.lock();
1334
1335            if !*started {
1336                break;
