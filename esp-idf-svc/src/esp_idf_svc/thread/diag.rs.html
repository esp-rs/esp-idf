30/// An entry of the Thread neighbor table (`otNeighborInfo`).
31///
32/// All fields describe the *neighboring* node, as observed by this node.
33#[derive(Copy, Clone, Debug, Default, Eq, PartialEq, Hash)]
34pub struct NeighborInfo {
35    /// IEEE 802.15.4 extended address of the neighbor, big-endian.
36    pub ext_address: u64,
37    /// Seconds since a frame was last received from the neighbor.
38    pub age: u32,
39    /// RLOC16 of the neighbor.
40    pub rloc16: u16,
41    /// Last received MAC frame counter of the neighbor.
42    pub link_frame_counter: u32,
43    /// Last received MLE frame counter of the neighbor.
44    pub mle_frame_counter: u32,
45    /// Link Quality In for messages from the neighbor (0..=3).
46    pub link_quality_in: u8,
47    /// Average RSSI (dBm) of frames from the neighbor, if available.
48    pub average_rssi: Option<i8>,
49    /// RSSI (dBm) of the last frame from the neighbor (clamped to <= 0), if available.
50    pub last_rssi: Option<i8>,
51    /// Frame error rate as a percentage (0..=100).
52    pub frame_error_rate: u8,
53    /// Message error rate as a percentage (0..=100).
54    pub message_error_rate: u8,
55    /// Whether the neighbor keeps its receiver on when idle.
56    pub rx_on_when_idle: bool,
57    /// Whether the neighbor is a Full Thread Device.
58    pub full_thread_device: bool,
59    /// Whether the neighbor requests the full Network Data.
60    pub full_network_data: bool,
61    /// Whether the neighbor is a child of this node.
62    pub is_child: bool,
63}
64
65impl NeighborInfo {
66    fn load_raw(raw: &otNeighborInfo) -> Self {
67        // OpenThread reports error rates as a fraction of 0xffff
68        let to_percent = |rate: u16| ((rate as u32 * 100) / 0xffff) as u8;
69        let rssi = |v: i8| (v != OT_RADIO_RSSI_INVALID as i8).then_some(v);
70
71        Self {
72            ext_address: u64::from_be_bytes(raw.mExtAddress.m8),
73            age: raw.mAge,
74            rloc16: raw.mRloc16,
75            link_frame_counter: raw.mLinkFrameCounter,
76            mle_frame_counter: raw.mMleFrameCounter,
77            link_quality_in: raw.mLinkQualityIn,
78            average_rssi: rssi(raw.mAverageRssi),
79            last_rssi: rssi(raw.mLastRssi).map(|v| v.min(0)),
80            frame_error_rate: to_percent(raw.mFrameErrorRate),
81            message_error_rate: to_percent(raw.mMessageErrorRate),
82            rx_on_when_idle: raw.mRxOnWhenIdle(),
83            full_thread_device: raw.mFullThreadDevice(),
84            full_network_data: raw.mFullNetworkData(),
85            is_child: raw.mIsChild(),
86        }
87    }
88}

486    /// Call the provided closure for each entry of the neighbor table.
487    pub fn neighbor_table<F>(&self, mut f: F) -> Result<(), EspError>
488    where
489        F: FnMut(&NeighborInfo),
490    {
491        let _lock = self.inner();
492
493        let instance = unsafe { esp_openthread_get_instance() };
494
495        let mut iter = OT_NEIGHBOR_INFO_ITERATOR_INIT as otNeighborInfoIterator;
496        let mut info = MaybeUninit::<otNeighborInfo>::uninit();
497
498        while unsafe { otThreadGetNextNeighborInfo(instance, &mut iter, info.as_mut_ptr()) }
499            == otError_OT_ERROR_NONE
500        {
501            f(&NeighborInfo::load_raw(unsafe { info.assume_init_ref() }));
502        }
503
504        Ok(())
505    }

507    /// Call the provided closure for each entry of the route table.
508    ///
509    /// On a Full Thread Device, this is the router table. On a Minimal Thread Device,
510    /// it is the parent of the node, if it has one.
511    pub fn route_table<F>(&self, mut f: F) -> Result<(), EspError>
512    where
513        F: FnMut(&RouteInfo),
514    {
515        let _lock = self.inner();
516
517        let instance = unsafe { esp_openthread_get_instance() };
518
519        let mut info = MaybeUninit::<otRouterInfo>::uninit();
520
521        #[cfg(esp_idf_openthread_ftd)]
522        {
523            let max_router_id = unsafe { otThreadGetMaxRouterId(instance) };
524
525            for router_id in 0..=max_router_id {
526                // Unallocated router IDs return an error
527                if unsafe { otThreadGetRouterInfo(instance, router_id as _, info.as_mut_ptr()) }
528                    == otError_OT_ERROR_NONE
529                {
530                    f(&RouteInfo::load_raw(unsafe { info.assume_init_ref() }));
531                }
532            }
533        }
534
535        #[cfg(not(esp_idf_openthread_ftd))]
536        {
537            if unsafe { otThreadGetParentInfo(instance, info.as_mut_ptr()) }
538                == otError_OT_ERROR_NONE
539            {
540                f(&RouteInfo::load_raw(unsafe { info.assume_init_ref() }));
541            }
542        }
543
544        Ok(())
545    }

547    /// Return the diagnostic-relevant parts of the Active Operational Dataset,
548    /// or `None` if the node has no Active Operational Dataset.
549    pub fn active_dataset_diag(&self) -> Result<Option<ActiveDatasetDiag>, EspError> {
550        let _lock = self.inner();
551
552        let mut dataset = MaybeUninit::<otOperationalDataset>::uninit();
553
554        if unsafe { otDatasetGetActive(esp_openthread_get_instance(), dataset.as_mut_ptr()) }
555            != otError_OT_ERROR_NONE
556        {
557            return Ok(None);
558        }
559
560        let dataset = unsafe { dataset.assume_init_ref() };
561
562        let components = OperationalDatasetComponents::load_raw(&dataset.mComponents);
563
564        Ok(Some(ActiveDatasetDiag {
565            active_timestamp: components
566                .active_timestamp_present
567                .then(|| (&dataset.mActiveTimestamp).into()),
568            security_policy: components
569                .security_policy_present
570                .then(|| SecurityPolicy::load_raw(&dataset.mSecurityPolicy)),
571            channel_mask: components
572                .channel_mask_present
573                .then_some(dataset.mChannelMask),
574            components,
575        }))
576    }

578    /// Return the diagnostic-relevant parts of the Pending Operational Dataset,
579    /// or `None` if the node has no Pending Operational Dataset.
580    pub fn pending_dataset_diag(&self) -> Result<Option<PendingDatasetDiag>, EspError> {
581        let _lock = self.inner();
582
583        let mut dataset = MaybeUninit::<otOperationalDataset>::uninit();
584
585        if unsafe { otDatasetGetPending(esp_openthread_get_instance(), dataset.as_mut_ptr()) }
586            != otError_OT_ERROR_NONE
587        {
588            return Ok(None);
589        }
590
591        let dataset = unsafe { dataset.assume_init_ref() };
592        let components = &dataset.mComponents;
593
594        Ok(Some(PendingDatasetDiag {
595            pending_timestamp: components
596                .mIsPendingTimestampPresent
597                .then(|| (&dataset.mPendingTimestamp).into()),
598            delay: components.mIsDelayPresent.then_some(dataset.mDelay),
599        }))
600    }
601
602    /// Return the Thread MLE counters.
603    pub fn mle_counters(&self) -> Result<MleCounters, EspError> {
604        let _lock = self.inner();
605
606        Ok(unsafe { &*otThreadGetMleCounters(esp_openthread_get_instance()) }.into())
607    }
608
609    /// Reset the Thread MLE counters.
610    pub fn reset_mle_counters(&self) -> Result<(), EspError> {
611        let _lock = self.inner();
612
613        unsafe { otThreadResetMleCounters(esp_openthread_get_instance()) };
614
615        Ok(())
616    }

619    pub fn mac_counters(&self) -> Result<MacCounters, EspError> {

621
622        Ok(unsafe { &*otLinkGetCounters(esp_openthread_get_instance()) }.into())
623    }

628
629        unsafe { otLinkResetCounters(esp_openthread_get_instance()) };
630
631        Ok(())
632    }
