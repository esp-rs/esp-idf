44/// An enum describing the status of either a concrete SRP service, or the SRP host.
45#[derive(Debug, Copy, Clone, Eq, PartialEq, Hash)]
46pub enum SrpState {
47    /// The service/host is to be added/registered.
48    ToAdd,
49    /// The service/host is being added/registered.
50    Adding,
51    /// The service/host is to be refreshed (re-register to renew lease).
52    ToRefresh,
53    /// The service/host is being refreshed.
54    Refreshing,
55    /// The service/host is to be removed/unregistered.
56    ToRemove,
57    /// The service/host is being removed/unregistered.
58    Removing,
59    /// The service/host has been removed/unregistered.
60    Removed,
61    /// The service/host is registered.
62    Registered,
63    /// Any other state.
64    Other(otSrpClientItemState),
65}
66
67impl Display for SrpState {
68    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
69        match self {
70            Self::ToAdd => write!(f, "To add"),
71            Self::Adding => write!(f, "Adding"),
72            Self::ToRefresh => write!(f, "To refresh"),
73            Self::Refreshing => write!(f, "Refreshing"),
74            Self::ToRemove => write!(f, "To remove"),
75            Self::Removing => write!(f, "Removing"),
76            Self::Removed => write!(f, "Removed"),
77            Self::Registered => write!(f, "Registered"),
78            Self::Other(state) => write!(f, "Other ({state})"),
79        }
80    }
81}

101/// The SRP configuration of the OpenThread stack.
102#[derive(Debug, Clone, Eq, PartialEq, Hash)]
103pub struct SrpConf<'a> {
104    /// SRP hostname
105    pub host_name: &'a str,
106    /// SRP host Ipv6 addresses.
107    /// If empty, the SRP implementation will automatically set the host addresses
108    /// by itself, using non-link-local addresses, once these become available.
109    pub host_addrs: &'a [Ipv6Addr],
110    /// SRP TTL (Time To Live) value.
111    pub ttl: u32,
112    /// Default lease time for SRP services if they specify 0 for their lease time.
113    /// Set to 0 to use the OpenThread default value.
114    pub default_lease_secs: u32,
115    /// Default key lease time for SRP services' keys if they specify 0 for their key lease time.
116    /// Set to 0 to use the OpenThread default value.
117    pub default_key_lease_secs: u32,
118}
119
120impl SrpConf<'_> {
121    /// Create a new `SrpConf` instance, wuth a host named "ot-device",
122    /// no explicit host addresses, a TTL of 60 seconds, and default lease times.
123    pub const fn new() -> Self {
124        Self {
125            host_name: "ot-device",
126            host_addrs: &[],
127            ttl: 60,
128            default_lease_secs: 0,
129            default_key_lease_secs: 0,
130        }
131    }
132
133    fn store(&self, ot_srp: &mut otSrpClientHostInfo, buf: &mut [u8]) -> Result<(), EspError> {
134        let (addrs, buf) = align_min::<otIp6Address>(buf, self.host_addrs.len())?;
135
136        ot_srp.mName = store_str(self.host_name, buf)?.0.as_ptr();
137
138        for (index, ip) in self.host_addrs.iter().enumerate() {
139            let addr = &mut addrs[index];
140            addr.mFields.m8 = ip.octets();
141        }
142
143        ot_srp.mAddresses = if addrs.is_empty() {
144            core::ptr::null_mut()
145        } else {
146            addrs.as_ptr()
147        };
148        ot_srp.mNumAddresses = addrs.len() as _;
149
150        #[cfg(not(esp_idf_version_major = "4"))]
151        {
152            ot_srp.mAutoAddress = addrs.is_empty();
153        }
154
155        Ok(())
156    }
157}

165/// An SRP service that can be registered with the OpenThread stack.
166#[derive(Debug, Clone, Eq, PartialEq, Hash)]
167pub struct SrpService<'a, SI, TI> {
168    /// The service name.
169    pub name: &'a str,
170    /// The instance name.
171    pub instance_name: &'a str,
172    /// The subtype labels.
173    pub subtype_labels: SI,
174    /// The TXT entries.
175    pub txt_entries: TI,
176    /// The service port.
177    pub port: u16,
178    /// The service priority.
179    pub priority: u16,
180    /// The service weight.
181    pub weight: u16,
182    /// The service lease time in seconds.
183    /// Set to 0 to use the default value as specified in `SrpConf`.
184    pub lease_secs: u32,
185    /// The service key lease time in seconds.
186    /// Set to 0 to use the default value as specified in `SrpConf`.
187    pub key_lease_secs: u32,
188}
189
190impl<'a, SI, TI> SrpService<'a, SI, TI>
191where
192    SI: Iterator<Item = &'a str> + Clone + 'a,
193    TI: Iterator<Item = (&'a str, &'a [u8])> + Clone + 'a,
194{
195    fn store(&self, ot_srp: &mut otSrpClientService, buf: &mut [u8]) -> Result<(), EspError> {
196        let subtype_labels_len = self.subtype_labels.clone().count();
197        let txt_entries_len = self.txt_entries.clone().count();
198
199        let (txt_entries, buf) = align_min::<otDnsTxtEntry>(buf, txt_entries_len)?;
200        let (subtype_labels, buf) = align_min::<*const char>(buf, subtype_labels_len + 1)?;
201
202        let (name, buf) = store_str(self.name, buf)?;
203        let (instance_name, buf) = store_str(self.instance_name, buf)?;
204
205        ot_srp.mName = name.as_ptr();
206        ot_srp.mInstanceName = instance_name.as_ptr();
207
208        let mut index = 0;
209        let mut buf = buf;
210
211        for subtype_label in self.subtype_labels.clone() {
212            let (subtype_label, rem_buf) = store_str(subtype_label, buf)?;
213
214            subtype_labels[index] = subtype_label.as_ptr() as *const _;
215
216            buf = rem_buf;
217            index += 1;
218        }
219
220        subtype_labels[index] = core::ptr::null();
221
222        index = 0;
223
224        for (key, value) in self.txt_entries.clone() {
225            let txt_entry = &mut txt_entries[index];
226
227            let (key, rem_buf) = store_str(key, buf)?;
228            let (value, rem_buf) = store_data(value, rem_buf)?;
229
230            txt_entry.mKey = key.as_ptr();
231            txt_entry.mValue = value.as_ptr();
232            txt_entry.mValueLength = value.len() as _;
233
234            buf = rem_buf;
235            index += 1;
236        }
237
238        ot_srp.mSubTypeLabels = subtype_labels.as_ptr() as *const _;
239        ot_srp.mTxtEntries = txt_entries.as_ptr();
240        ot_srp.mNumTxtEntries = txt_entries_len as _;
241        ot_srp.mPort = self.port;
242        ot_srp.mPriority = self.priority;
243        ot_srp.mWeight = self.weight;
244        #[cfg(not(esp_idf_version_major = "4"))]
245        {
246            ot_srp.mLease = self.lease_secs;
247            ot_srp.mKeyLease = self.key_lease_secs;
248        }
249        ot_srp.mState = 0;
250        ot_srp.mNext = core::ptr::null_mut();
251
252        Ok(())
253    }
254}

267            self.port,
268            self.priority,

285            if index > 0 {
286                write!(f, ", {}: {:?}", value.0, value.1)?;
287            } else {
288                write!(f, "{}: {:?}", value.0, value.1)?;
289            }
290        }
291
292        write!(f, "]}}")
293    }

300/// An iterator over the subtype labels of an SRP service
301/// as returned by `OpenThread::srp_services`.
302#[derive(Clone)]
303pub struct OutSrpSubtypeLabelsIter<'a> {
304    ptr: *const *const u8,
305    index: usize,
306    _phantom: PhantomData<&'a ()>,
307}
308
309impl<'a> Iterator for OutSrpSubtypeLabelsIter<'a> {
310    type Item = &'a str;
311
312    fn next(&mut self) -> Option<Self::Item> {
313        if self.ptr.is_null() {
314            return None;
315        }
316
317        let label = unsafe { *self.ptr.add(self.index) };
318
319        if label.is_null() {
320            None
321        } else {
322            self.index += 1;
323            Some(unsafe { CStr::from_ptr(label as _) }.to_str().unwrap())
324        }
325    }
326}

328/// An iterator over the TXT entries of an SRP service
329/// as returned by `OpenThread::srp_services`.
330#[derive(Clone)]
331pub struct OutSrpTxtEntriesIter<'a> {
332    ptr: *const otDnsTxtEntry,
333    size: usize,
334    index: usize,
335    _phantom: PhantomData<&'a ()>,
336}
337
338impl<'a> Iterator for OutSrpTxtEntriesIter<'a> {
339    type Item = (&'a str, &'a [u8]);
340
341    fn next(&mut self) -> Option<Self::Item> {
342        if self.ptr.is_null() || self.index == self.size {
343            return None;
344        }
345
346        let entry = unsafe { self.ptr.add(self.index) };
347
348        self.index += 1;
349
350        let entry = unsafe { &*entry };
351
352        Some((
353            unsafe { CStr::from_ptr(entry.mKey) }.to_str().unwrap(),
354            unsafe { core::slice::from_raw_parts(entry.mValue, entry.mValueLength as _) },
355        ))
356    }
357}

407    /// Return the current SRP client configuration and SRP client host state to the provided closure.
408    ///
409    /// Arguments:
410    /// - `f`: A closure that takes the SRP configuration and SRP host state as arguments.
411    pub fn srp_conf<F, R>(&self, f: F) -> Result<R, EspError>
412    where
413        F: FnOnce(&SrpConf, SrpState, bool) -> Result<R, EspError>,
414    {
415        let inner = self.inner();
416
417        let instance = unsafe { esp_openthread_get_instance() };
418
419        let info = unsafe { otSrpClientGetHostInfo(instance).as_ref() }.unwrap();
420
421        #[allow(unused_mut)]
422        let mut conf = SrpConf {
423            host_name: if !info.mName.is_null() {
424                unsafe { CStr::from_ptr(info.mName) }.to_str().unwrap()
425            } else {
426                ""
427            },
428            host_addrs: if info.mNumAddresses > 0 && !info.mAddresses.is_null() {
429                unsafe {
430                    core::slice::from_raw_parts(
431                        info.mAddresses as *const _,
432                        info.mNumAddresses as _,
433                    )
434                }
435            } else {
436                &[]
437            },
438            ttl: 0,
439            default_lease_secs: 0,
440            default_key_lease_secs: 0,
441        };
442
443        #[cfg(not(esp_idf_version_major = "4"))]
444        {
445            unsafe {
446                conf.ttl = otSrpClientGetTtl(instance);
447                conf.default_lease_secs = otSrpClientGetLeaseInterval(instance);
448                conf.default_key_lease_secs = otSrpClientGetKeyLeaseInterval(instance);
449            }
450        }
451
452        f(&conf, info.mState.into(), !inner.srp.conf_taken)
453    }

456    pub fn srp_is_empty(&self) -> Result<bool, EspError> {
457        let inner = self.inner();

462    /// Set the SRP client configuration.
463    ///
464    /// Arguments:
465    /// - `conf`: The SRP configuration.
466    ///
467    /// Returns:
468    /// - `Ok(())` if the configuration was set successfully.
469    /// - `Err(OtError)` if the configuration could not be set. One reason why the configuration setting
470    ///   might fail is if the configuration had already been set and then not removed with `srp_remove_all`.
471    pub fn srp_set_conf(&self, conf: &SrpConf) -> Result<(), EspError> {
472        let mut inner = self.inner();
473
474        let instance = unsafe { esp_openthread_get_instance() };
475
476        if inner.srp.conf_taken {
477            esp!(ESP_ERR_INVALID_STATE)?;
478        }
479
480        #[cfg(not(esp_idf_version_major = "4"))]
481        unsafe {
482            otSrpClientSetLeaseInterval(instance, conf.default_lease_secs);
483            otSrpClientSetKeyLeaseInterval(instance, conf.default_key_lease_secs);
484            otSrpClientSetTtl(instance, conf.ttl);
485        }
486
487        let mut srp_conf = otSrpClientHostInfo {
488            mName: core::ptr::null(),
489            mAddresses: core::ptr::null(),
490            mNumAddresses: 0,
491            #[cfg(not(esp_idf_version_major = "4"))]
492            mAutoAddress: true,
493            mState: 0,
494        };
495
496        conf.store(&mut srp_conf, &mut inner.srp.conf_buf)?;
497        inner.srp.conf_taken = true;
498
499        ot_esp!(unsafe { otSrpClientSetHostName(instance, srp_conf.mName) })?;
500
501        if !conf.host_addrs.is_empty() {
502            ot_esp!(unsafe {
503                otSrpClientSetHostAddresses(instance, srp_conf.mAddresses, srp_conf.mNumAddresses)
504            })?;
505        } else {
506            #[cfg(not(esp_idf_version_major = "4"))]
507            {
508                ot_esp!(unsafe { otSrpClientEnableAutoHostAddress(instance) })?;
509            }
510        }
511
512        Ok(())
513    }

595    /// Iterate over the SRP services registered with the SRP client.
596    ///
597    /// Arguments:
598    /// - `f`: A closure that receives a tuple of the next SRP service, SRP service state, and SRP service ID.
599    ///   If there are no more SRP services, the closure will receive `None`.
600    pub fn srp_services<F>(&self, mut f: F) -> Result<(), EspError>
601    where
602        F: FnMut(Option<(&OutSrpService<'_>, SrpState, SrpServiceSlot)>),
603    {
604        let inner = self.inner();
605
606        let mut service_ptr: *const otSrpClientService =
607            unsafe { otSrpClientGetServices(esp_openthread_get_instance()) };
608
609        while !service_ptr.is_null() {
610            let service = unsafe { &*service_ptr };
611
612            let slot = inner
613                .srp
614                .services
615                .iter()
616                .position(|s| core::ptr::eq(&s.service, service))
617                .unwrap();
618
619            f(Some((&service.into(), service.mState.into(), slot)));
620
621            service_ptr = service.mNext;
622        }
623
624        f(None);
625
626        Ok(())
627    }

630    ///
631    /// Arguments:
632    /// - `service`: The SRP service to add.
633    ///
634    /// Returns:
635    /// - The SRP service slot, if the service was added successfully.
636    /// - `Err(OtError)` if the service could not be added. One reason why the service addition
637    ///   might fail is if there are no more slots available for services. This can happen even if all services
638    ///   had been removed, as the slots are not freed until the SRP client propagates the removal info to the SRP server.
639    pub fn srp_add_service<'a, SI, TI>(
640        &self,
641        service: &'a SrpService<'a, SI, TI>,
642    ) -> Result<SrpServiceSlot, EspError>
643    where
644        SI: Iterator<Item = &'a str> + Clone + 'a,
645        TI: Iterator<Item = (&'a str, &'a [u8])> + Clone + 'a,
646    {
647        let mut inner = self.inner();
648
649        let slot = inner.srp.services.iter().position(|service| !service.taken);
650
651        let Some(slot) = slot else {
652            return Err(ot_esp_err(otError_OT_ERROR_NO_BUFS));
653        };
654
655        let our_service = &mut inner.srp.services[slot];
656
657        service.store(&mut our_service.service, &mut our_service.buf)?;
658
659        ot_esp!(unsafe {
660            otSrpClientAddService(esp_openthread_get_instance(), &mut our_service.service)
661        })?;
662
663        debug!("Service added");
664
665        our_service.taken = true;
666
667        Ok(slot)
668    }
669
670    /// Remove an SRP service from the SRP client.
671    ///
672    /// Arguments:
673    /// - `slot`: The SRP service to remove.
674    /// - `immediate`: If `true`, the service will be removed immediately, otherwise, the service will be removed gracefully
675    ///   by propagating the removal info to the SRP server.
676    pub fn srp_remove_service(
677        &self,
678        slot: SrpServiceSlot,
679        immediate: bool,
680    ) -> Result<(), EspError> {
681        let mut inner = self.inner();
682
683        if slot >= inner.srp.services.len() || !inner.srp.services[slot].taken {
684            ot_esp!(otError_OT_ERROR_INVALID_ARGS)?;
685        }
686
687        let service = &mut inner.srp.services[slot];
688
689        if immediate {
690            ot_esp!(unsafe {
691                otSrpClientClearService(esp_openthread_get_instance(), &mut service.service)
692            })?;
693            service.taken = false;
694            debug!("Service {slot} cleared immeidately");
695        } else {
696            ot_esp!(unsafe {
697                otSrpClientRemoveService(esp_openthread_get_instance(), &mut service.service)
698            })?;
699            debug!("Service {slot} scheduled for removal");
700        }
701
702        Ok(())
703    }

705    /// Remove the SRP hostname and all SRP services from the SRP client.
706    ///
707    /// Arguments:
708    /// - `immediate`: If `true`, the hostname and services will be removed immediately, otherwise,
709    ///   the hostname and services will be removed gracefully by propagating the removal info to the SRP server.
710    pub fn srp_remove_all(&self, immediate: bool) -> Result<(), EspError> {
711        let mut inner = self.inner();
712
713        let instance = unsafe { esp_openthread_get_instance() };
714
715        if immediate {
716            unsafe {
717                otSrpClientClearHostAndServices(instance);
718            }
719
720            inner.srp.conf_taken = false;
721            for service in &mut inner.srp.services {
722                service.taken = false;
723            }
724
725            debug!("Hostname and all services cleared immediately");
726        } else {
727            ot_esp!(unsafe { otSrpClientRemoveHostAndServices(instance, false, true) })?;
728            debug!("Hostname and all services scheduled for removal");
729        }
730
731        Ok(())
732    }

912pub(crate) struct OtSrp {
913    conf_taken: bool,
914    conf_buf: [u8; SRP_HOST_BUF_SIZE],
915    services: [OtSrpService; SRP_SVCS],
916    changed: Arc<Notification>,
917}

919impl OtSrp {
920    pub(crate) unsafe fn init(this: *mut Self) {
921        unsafe {
922            addr_of_mut!((*this).conf_taken).write(false);
923            addr_of_mut!((*this).conf_buf).write_bytes(0, 1);
924            addr_of_mut!((*this).changed).write(Arc::new(Notification::new()));
925
926            for index in 0..SRP_SVCS {
927                let service = addr_of_mut!((*this).services[index]);
928                OtSrpService::init(service);
929            }
930        }
931    }
932
933    /// Reclaims the slots of the SRP host and services that are reported as removed
934    fn cleanup(
935        &mut self,
936        host_info: &otSrpClientHostInfo,
937        mut removed_services: Option<&otSrpClientService>,
938    ) {
939        if host_info.mState == otSrpClientItemState_OT_SRP_CLIENT_ITEM_STATE_REMOVED {
940            self.conf_taken = false;
941            info!("SRP host removed");
942        }
943
944        while let Some(service) = removed_services {
945            let (slot, our_service) = self
946                .services
947                .iter_mut()
948                .enumerate()
949                .find(|(_, s)| core::ptr::eq(&s.service, service))
950                .unwrap();
951
952            removed_services = unsafe { service.mNext.as_ref() };
953
954            our_service.taken = false;
955            info!("SRP service at slot {slot} removed");
956        }
957    }
958
959    fn plat_srp_changed(
960        &mut self,
961        error: otError,
962        host_info: &otSrpClientHostInfo,
963        _services: Option<&otSrpClientService>,
964        removed_services: Option<&otSrpClientService>,
965    ) {
966        trace!("Plat SRP changed callback");
967
968        if error != otError_OT_ERROR_NONE {
969            // The SRP client keeps retrying on its own, but without this the failure
970            // is completely invisible (i.e. services silently never get published)
971            let reason = unsafe { CStr::from_ptr(otThreadErrorToString(error)) }
972                .to_str()
973                .unwrap_or("Unknown");
974
975            warn!(
976                "SRP update failed: {reason} ({error}); host is {}",
977                SrpState::from(host_info.mState)
978            );
979
980            if error == otError_OT_ERROR_DUPLICATED {
981                // The SRP server maps this from a `YXDOMAIN` response, which it returns
982                // when the host name - or one of the service instance names - is already
983                // registered there under a *different* ECDSA key. The client cannot
984                // recover on its own: it will keep retrying until the stale registration
985                // reaches the end of its key lease (14 days by default) or the server is
986                // told to drop it.
987                warn!(
988                    "SRP name is registered on the server under a different key; \
989                     the SRP key of this device changed, or another device claimed the name"
990                );
991            }
992        }
993
994        self.cleanup(host_info, removed_services);
995
996        self.changed.notify_lsb();
997    }
998
999    fn plat_srp_auto_started(&mut self) {
1000        self.changed.notify_lsb();
1001    }

1004        error: otError,
1005        host_info: *const crate::sys::otSrpClientHostInfo,
1006        services: *const crate::sys::otSrpClientService,
1007        removed_services: *const crate::sys::otSrpClientService,

1009    ) {
1010        let srp = context as *mut OtSrp;
1011        let srp = unsafe { srp.as_mut() }.unwrap();

1013        srp.plat_srp_changed(

1016            unsafe { services.as_ref() },
1017            unsafe { removed_services.as_ref() },
1018        );
1019    }

1048fn align_min<T>(buf: &mut [u8], count: usize) -> Result<(&mut [T], &mut [u8]), EspError> {
1049    if count == 0 || core::mem::size_of::<T>() == 0 {
1050        return Ok((&mut [], buf));
1051    }
1052
1053    let (t_leading_buf0, t_buf, _) = unsafe { buf.align_to_mut::<T>() };
1054    if t_buf.len() < count {
1055        ot_esp!(otError_OT_ERROR_NO_BUFS)?;
1056    }
1057
1058    // Shrink `t_buf` to the number of requested items (count)
1059    let t_buf = &mut t_buf[..count];
1060    let t_leading_buf0_len = t_leading_buf0.len();
1061    let t_buf_size = core::mem::size_of_val(t_buf);
1062
1063    let (buf0, remaining_buf) = buf.split_at_mut(t_leading_buf0_len + t_buf_size);
1064
1065    let (t_leading_buf, t_buf, t_remaining_buf) = unsafe { buf0.align_to_mut::<T>() };
1066    assert_eq!(t_leading_buf0_len, t_leading_buf.len());
1067    assert_eq!(t_buf.len(), count);
1068    assert!(t_remaining_buf.is_empty());
1069
1070    Ok((t_buf, remaining_buf))
1071}

1073fn store_str<'t>(str: &str, buf: &'t mut [u8]) -> Result<(&'t CStr, &'t mut [u8]), EspError> {
1074    let data_len = str.len() + 1;
1075
1076    if data_len > buf.len() {
1077        ot_esp!(otError_OT_ERROR_NO_BUFS)?;
1078    }
1079
1080    let (str_buf, rem_buf) = buf.split_at_mut(data_len);
1081
1082    str_buf[..str.len()].copy_from_slice(str.as_bytes());
1083    str_buf[str.len()] = 0;
1084
1085    Ok((
1086        CStr::from_bytes_with_nul(&str_buf[..data_len]).unwrap(),
1087        rem_buf,
1088    ))
1089}

1091fn store_data<'t>(data: &[u8], buf: &'t mut [u8]) -> Result<(&'t [u8], &'t mut [u8]), EspError> {
1092    if data.len() > buf.len() {
1093        ot_esp!(otError_OT_ERROR_NO_BUFS)?;
1094    }
1095
1096    let (data_buf, rem_buf) = buf.split_at_mut(data.len());
1097
1098    data_buf[..data.len()].copy_from_slice(data);
1099
1100    Ok((data_buf, rem_buf))
1101}
