8impl From<ipv4::Ipv4Addr> for Newtype<esp_ip4_addr_t> {

22// lwIP's own IPv4 address type only exists with IPv4 enabled (`CONFIG_LWIP_IPV4` only exists
23// since ESP-IDF 5.1; before that IPv4 is always enabled)
24#[cfg(any(esp_idf_lwip_ipv4, not(esp_idf_version_at_least_5_1_0)))]
25impl From<ipv4::Ipv4Addr> for Newtype<ip4_addr_t> {
26    fn from(ip: ipv4::Ipv4Addr) -> Self {
27        let result: Newtype<esp_ip4_addr_t> = ip.into();
28
29        Newtype(ip4_addr_t {
30            addr: result.0.addr,

32    }
33}

106impl From<Newtype<esp_ip6_addr_t>> for core::net::Ipv6Addr {
107    fn from(value: Newtype<esp_ip6_addr_t>) -> Self {
108        let mut out = [0; 16];
109        value
110            .0
111            .addr
112            .into_iter()
113            .map(u32::from_be)
114            .flat_map(u32::to_be_bytes)
115            .zip(out.iter_mut())
116            .for_each(|(i, o)| *o = i);
117
118        core::net::Ipv6Addr::from(out)
119    }
