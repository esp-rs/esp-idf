1//! RMT (Remote Control) peripheral support.
2//!
3//! The RMT (Remote Control Transceiver) peripheral was designed to act as an infrared transceiver.
4//! However, due to the flexibility of its data format, RMT can be extended to a versatile and
5//! general-purpose transceiver, transmitting or receiving many other types of signals.
6//! From the perspective of network layering, the RMT hardware contains both physical and data
7//! link layers. The physical layer defines the communication media and bit signal representation.
8//! The data link layer defines the format of an RMT frame. The minimal data unit in the frame is
9//! called the RMT symbol, which is represented by [`Symbol`] in the driver.
10//!
11//! ESP32 contains multiple channels in the RMT peripheral. Each channel can be independently
12//! configured as either transmitter or receiver.
13//!
14//! Typically, the RMT peripheral can be used in the following scenarios:
15//! - Transmit or receive infrared signals, with any IR protocols, e.g., NEC
16//! - General-purpose sequence generator
17//! - Transmit signals in a hardware-controlled loop, with a finite or infinite number of times
18//! - Multi-channel simultaneous transmission
19//! - Modulate the carrier to the output signal or demodulate the carrier from the input signal
20//!
21//! # Driver redesign in ESP-IDF 5.0
22//!
23//! In ESP-IDF 5.0, the [RMT API was redesigned] to simplify and unify the usage of the
24//! RMT peripheral.
25//!
26//! It is recommended to use the new API, but for now the old API is available through
27//! the `rmt-legacy` feature. The ESP-IDF 6.0 release will remove support for the legacy API.
28//!
29//! [RMT API was redesigned]: https://docs.espressif.com/projects/esp-idf/en/stable/esp32/migration-guides/release-5.x/5.0/peripherals.html#rmt-driver

59/// A [`Symbol`] constists of two [`Pulse`]s, where each pulse defines a level of the pin (high or low)
60/// and a duration ([`PulseTicks`]).
61///
62/// The `Pulse`s can be the same level (e.g., both high) and must not necessarily be different levels.
63///
64/// This is just a newtype over the IDF's `rmt_item32_t` or `rmt_symbol_word_t` type.
65#[derive(Clone, Copy)]
66#[repr(transparent)]
67pub struct Symbol(rmt_symbol_word_t);
68
69impl Symbol {
70    /// Create a symbol from a pair of half-cycles.
71    #[must_use]
72    pub fn new(level0: Pulse, level1: Pulse) -> Self {
73        let item = rmt_symbol_word_t { val: 0 };
74        let mut this = Self(item);
75        this.update(level0, level1);
76        this
77    }
78
79    /// Constructs a symbol from the given levels and durations.
80    ///
81    /// This is a convenience function that combines [`Pulse::new_with_duration`] and [`Symbol::new`].
82    pub fn new_with(
83        resolution: Hertz,
84        level0: PinState,
85        duration0: Duration,
86        level1: PinState,
87        duration1: Duration,
88    ) -> Result<Self, EspError> {
89        Ok(Self::new(
90            Pulse::new_with_duration(resolution, level0, duration0)?,
91            Pulse::new_with_duration(resolution, level1, duration1)?,
92        ))
93    }
94
95    /// Constructs a new symbol where the duration is split evenly between the two levels.
96    ///
97    /// It is guaranteed that the combined duration of the two levels adds up to the given duration.
98    /// If the duration is odd, the first level might be a bit shorter than the second level.
99    pub fn new_half_split(
100        resolution: Hertz,
101        level0: PinState,
102        level1: PinState,
103        duration: Duration,
104    ) -> Result<Self, EspError> {
105        let first_half_duration = duration / 2;
106        // This ensures that the two halves always add up to the original duration,
107        // even if the duration is odd.
108        let second_half_duration = duration - first_half_duration;
109
110        Ok(Self::new(
111            Pulse::new_with_duration(resolution, level0, first_half_duration)?,
112            Pulse::new_with_duration(resolution, level1, second_half_duration)?,
113        ))
114    }
115
116    /// Repeats the symbol to have the returned sequence of symbols last for
117    /// exactly the given duration.
118    ///
119    /// There is an upper limit for how long a single symbol can be ([`PulseTicks::max`]).
120    /// To create a symbol that lasts longer than that, it is split into multiple symbols.
121    /// The returned iterator yields as many symbols as necessary to reach the
122    /// desired duration.
123    ///
124    /// If the given duration is not a multiple of the symbol duration,
125    /// the last symbol will be adjusted to occupy the remaining time,
126    /// rounded down to whole ticks.
127    ///
128    /// # Panics
129    ///
130    /// If `self` has a duration of zero.
131    pub fn repeat_for(&self, resolution: Hertz, duration: Duration) -> impl Iterator<Item = Self> {
132        let ticks0 = self.level0().ticks.ticks() as u128;
133        let symbol_ticks = ticks0 + self.level1().ticks.ticks() as u128;
134
135        // Handle edge-case to prevent an infinite loop:
136        if symbol_ticks == 0 {
137            panic!("Cannot repeat a symbol with zero duration for {duration:?}");
138        }
139
140        let duration_ticks =
141            duration.as_nanos() * resolution.0 as u128 / Duration::from_secs(1).as_nanos();
142
143        let count = duration_ticks / symbol_ticks;
144        let remainder = duration_ticks % symbol_ticks;
145
146        let last_symbol = (remainder > 0).then(|| {
147            // `remainder < symbol_ticks`, so both halves fit into their `PulseTicks`
148            let remainder0 = ticks0.min(remainder);
149            let remainder1 = remainder - remainder0;
150
151            Self::new(
152                Pulse::new(
153                    self.level0().pin_state,
154                    PulseTicks::new(remainder0 as u16).unwrap(),
155                ),
156                Pulse::new(
157                    self.level1().pin_state,
158                    PulseTicks::new(remainder1 as u16).unwrap(),
159                ),
160            )
161        });
162
163        core::iter::repeat_n(*self, count as usize).chain(core::iter::once(last_symbol).flatten())
164    }
165
166    #[must_use]
167    fn symbol_word_to_pulse(word: &rmt_symbol_word_t) -> (Pulse, Pulse) {
168        let inner = unsafe { &word.__bindgen_anon_1 };
169        (
170            Pulse::new(
171                (inner.level0() as u32).into(),
172                PulseTicks::new(inner.duration0()).unwrap(),
173            ),
174            Pulse::new(
175                (inner.level1() as u32).into(),
176                PulseTicks::new(inner.duration1()).unwrap(),
177            ),
178        )
179    }
180
181    /// Returns the first half-cycle (pulse) of this symbol.
182    #[must_use]
183    pub fn level0(&self) -> Pulse {
184        Self::symbol_word_to_pulse(&self.0).0
185    }
186
187    /// Returns the second half-cycle (pulse) of this symbol.
188    #[must_use]
189    pub fn level1(&self) -> Pulse {
190        Self::symbol_word_to_pulse(&self.0).1
191    }
192
193    /// Returns the combined duration of both half-cycles of this symbol.
194    pub fn duration(&self, resolution: Hertz) -> Duration {
195        self.level0().ticks.duration(resolution) + self.level1().ticks.duration(resolution)
196    }
197
198    /// Mutate this symbol to store a different pair of half-cycles.
199    pub fn update(&mut self, level0: Pulse, level1: Pulse) {
200        // SAFETY: We're overriding all 32 bits, so it doesn't matter what was here before.
201        let inner = unsafe { &mut self.0.__bindgen_anon_1 };
202        inner.set_level0(level0.pin_state as u16);
203        inner.set_duration0(level0.ticks.ticks());
204        inner.set_level1(level1.pin_state as u16);
205        inner.set_duration1(level1.ticks.ticks());
206    }
207}

274/// Type of RMT RX done event data.
275#[derive(Debug, Clone, Copy)]
276#[non_exhaustive]
277pub struct RxDoneEventData {
278    /// Pointer to the received RMT symbols.
279    ///
280    /// These symbols are saved in the internal `buffer` of the `RxChannelDriver`.
281    /// You are not allowed to free this buffer.
282    ///
283    /// If the partial receive feature is enabled, then the buffer will be
284    /// used as a "second level buffer", where its content can be overwritten by
285    /// data coming in afterwards. In this case, you should copy the received data to
286    /// another place if you want to keep it or process it later.
287    pub received_symbols: *mut Symbol,
288    /// The number of received RMT symbols.
289    ///
290    /// This value will never be larger than the buffer size of the buffer passed to the
291    /// receive function.
292    ///
293    /// If the buffer is not sufficient to accomodate all the received RMT symbols, the
294    /// driver only keeps the maximum number of symbols that the buffer can hold, and excess
295    /// symbols are discarded or ignored.
296    pub num_symbols: usize,
297    /// Indicates whether the current received data is the last part of the transaction.
298    ///
299    /// This is useful for when [`ReceiveConfig::enable_partial_rx`] is enabled,
300    /// and the data is received in parts.
301    ///
302    /// For receives where partial rx is not enabled, this field will always be `true`.
303    ///
304    /// [`ReceiveConfig::enable_partial_rx`]: crate::rmt::config::ReceiveConfig::enable_partial_rx
305    #[cfg(esp_idf_version_at_least_5_3_0)]
306    pub is_last: bool,
307}
308
309impl RxDoneEventData {
310    /// Returns the received symbols as a slice.
311    ///
312    /// # Safety
313    ///
314    /// Both the pointer and the length must be valid.
315    /// If these haven't been changed from the values provided by the driver,
316    /// this should be the case **in** the ISR callback.
317    pub unsafe fn as_slice(&self) -> &[Symbol] {
318        core::slice::from_raw_parts(self.received_symbols, self.num_symbols)
319    }
320}
321
322impl From<rmt_rx_done_event_data_t> for RxDoneEventData {
323    fn from(data: rmt_rx_done_event_data_t) -> Self {
324        Self {
325            received_symbols: data.received_symbols as *mut Symbol,
326            num_symbols: data.num_symbols,
327            #[cfg(esp_idf_version_at_least_5_3_0)]
328            is_last: data.flags.is_last() != 0,
329        }
330    }
331}

333/// A trait implemented by an RMT channel that provides common functionality.
334pub trait RmtChannel {
335    /// Returns the underlying `rmt_channel_handle_t`.
336    fn handle(&self) -> rmt_channel_handle_t;
337
338    /// Returns whether the channel is currently enabled.
339    #[must_use]
340    fn is_enabled(&self) -> bool;
341
342    /// Sets the internal `is_enabled` flag of the channel that is used to track the channel state.
343    ///
344    /// # Safety
345    ///
346    /// The channel assumes that the given `is_enabled` reflects the state of the channel.
347    /// There shouldn't be any reason for the user to call this function directly.
348    /// Use `enable`/`disable` instead.
349    #[doc(hidden)]
350    unsafe fn set_internal_enabled(&mut self, is_enabled: bool);
351
352    /// Must be called in advance before transmitting or receiving RMT symbols.
353    /// For TX channels, enabling a channel enables a specific interrupt and
354    /// prepares the hardware to dispatch transactions.
355    ///
356    /// For RX channels, enabling a channel enables an interrupt, but the receiver
357    /// is not started during this time, as the characteristics of the incoming
358    /// signal have yet to be specified.
359    ///
360    /// The receiver is started in [`RxChannel::receive`].
361    fn enable(&mut self) -> Result<(), EspError> {
362        esp!(unsafe { rmt_enable(self.handle()) })?;
363
364        // SAFETY: The channel has been enabled -> it is safe to mark it as enabled.
365        unsafe { self.set_internal_enabled(true) };
366        Ok(())
367    }
368
369    /// Disables the interrupt and clearing any pending interrupts. The transmitter
370    /// and receiver are disabled as well.
371    ///
372    /// # Note
373    ///
374    /// This function will release a power management (PM) lock that might be
375    /// installed during channel allocation
376    ///
377    /// # Errors
378    ///
379    /// - `ESP_ERR_INVALID_ARG`: Disable RMT channel failed because of invalid argument
380    /// - `ESP_ERR_INVALID_STATE`: Disable RMT channel failed because it's not enabled yet
381    /// - `ESP_FAIL`: Disable RMT channel failed because of other error
382    fn disable(&mut self) -> Result<(), EspError> {
383        esp!(unsafe { rmt_disable(self.handle()) })?;
384        // SAFETY: The channel has been disable -> it is safe to mark it as disabled.
385        unsafe { self.set_internal_enabled(false) };
386        Ok(())
387    }
388
389    /// Apply (de)modulation feature for the channel.
390    ///
391    /// If `carrier_config` is `None`, the carrier (de)modulation will be disabled.
392    ///
393    /// # Errors
394    ///
395    /// - `ESP_ERR_INVALID_ARG`: Apply carrier failed because of invalid argument
396    /// - `ESP_FAIL`: Apply carrier failed because of other error
397    fn apply_carrier(&mut self, carrier_config: Option<&CarrierConfig>) -> Result<(), EspError> {
398        let mut sys_carrier = None;
399        if let Some(CarrierConfig {
400            frequency,
401            duty_cycle,
402            polarity_active_low,
403            always_on,
404            // match __internal to get a compiler error if new fields are added in the future
405            __internal,
406        }) = carrier_config
407        {
408            sys_carrier = Some(rmt_carrier_config_t {
409                frequency_hz: (*frequency).into(),
410                duty_cycle: duty_cycle.to_f32(),
411                flags: rmt_carrier_config_t__bindgen_ty_1 {
412                    _bitfield_1: rmt_carrier_config_t__bindgen_ty_1::new_bitfield_1(
413                        *polarity_active_low as u32,
414                        *always_on as u32,
415                    ),
416                    ..Default::default()
417                },
418            })
419        }
420
421        esp!(unsafe {
422            rmt_apply_carrier(
423                self.handle(),
424                sys_carrier.as_ref().map_or(ptr::null(), |c| c as *const _),
425            )
426        })
427    }
428}

430impl<R: RmtChannel> RmtChannel for &mut R {
431    fn handle(&self) -> rmt_channel_handle_t {
432        (**self).handle()
433    }
434
435    fn is_enabled(&self) -> bool {
436        (**self).is_enabled()
437    }
438
439    unsafe fn set_internal_enabled(&mut self, is_enabled: bool) {
440        (**self).set_internal_enabled(is_enabled)

442
443    fn enable(&mut self) -> Result<(), EspError> {
444        (**self).enable()
445    }
446
447    fn disable(&mut self) -> Result<(), EspError> {
448        (**self).disable()
449    }
450
451    fn apply_carrier(&mut self, carrier_config: Option<&CarrierConfig>) -> Result<(), EspError> {
452        (**self).apply_carrier(carrier_config)
453    }
454}

456/// Clock source for RMT channels.
457#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Default)]
458pub enum ClockSource {
459    #[default]
460    Default,
461    #[cfg(any(esp32, esp32c3, esp32s2, esp32s3))]
462    APB,
463    #[cfg(any(
464        esp32c3, esp32c5, esp32c6, esp32h2, esp32h21, esp32h4, esp32p4, esp32s3, esp32s31
465    ))]
466    RcFast,
467    #[cfg(any(esp32, esp32s2))]
468    RefTick,
469    #[cfg(any(
470        esp32c3, esp32c5, esp32c6, esp32h2, esp32h21, esp32h4, esp32p4, esp32s3, esp32s31
471    ))]
472    XTAL,
473    #[cfg(any(esp32c5, esp32c6, esp32p4, esp32s31))]
474    PLLF80M,
475}

477impl From<ClockSource> for rmt_clock_source_t {
478    fn from(clock: ClockSource) -> Self {
479        match clock {
480            ClockSource::Default => soc_periph_rmt_clk_src_t_RMT_CLK_SRC_DEFAULT,
481            #[cfg(any(esp32, esp32c3, esp32s2, esp32s3))]
482            ClockSource::APB => soc_periph_rmt_clk_src_t_RMT_CLK_SRC_APB,
483            #[cfg(any(
484                esp32c3, esp32c5, esp32c6, esp32h2, esp32h21, esp32h4, esp32p4, esp32s3, esp32s31
485            ))]
486            ClockSource::RcFast => soc_periph_rmt_clk_src_t_RMT_CLK_SRC_RC_FAST,
487            #[cfg(any(esp32, esp32s2))]
488            ClockSource::RefTick => soc_periph_rmt_clk_src_t_RMT_CLK_SRC_REF_TICK,
489            #[cfg(any(
490                esp32c3, esp32c5, esp32c6, esp32h2, esp32h21, esp32h4, esp32p4, esp32s3, esp32s31
491            ))]
492            ClockSource::XTAL => soc_periph_rmt_clk_src_t_RMT_CLK_SRC_XTAL,
493            #[cfg(any(
494                esp32c5,
495                esp32c6,
496                esp32p4,
497                all(esp32s31, not(esp_idf_version_at_least_6_1_1))

499            ClockSource::PLLF80M => soc_periph_rmt_clk_src_t_RMT_CLK_SRC_PLL_F80M,

502            ClockSource::PLLF80M => soc_periph_rmt_clk_src_t_RMT_CLK_SRC_REF_F80M,
503        }
504    }
